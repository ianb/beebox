// # Print and eventually

import { test } from "tap";
import { eventually, printed } from "agent-doctest/check";

void test("line 3", async (t) => {
  const out = printed();
  const print = out.print;
  // doctest:L4
  print("before");
  await t.checkText(out.with(1 + 1), `before
2`);
});

void test("line 10", async (t) => {
  // doctest:L11
  let ready = false;
  setTimeout(() => { ready = true; }, 5);
  await t.checkText(await eventually(() => ready), `true`);
});
