// # Collisions
//
// Setup declares `t`, and an example declares `tap`, `out`, and `printed`: the
// converter's own names must not collide with them.

import tap3, { test } from "tap";
import { printed as printed2 } from "agent-doctest/check";
import type { Stats } from "node:fs";

const t = 1;

tap3.teardown(async () => {
  try {

    const stats: Stats | null = null;
    void stats;
  } catch (error) {
    console.error("teardown failed:", error);
    process.exitCode = 1;
  }
});

void test("line 16", async (tap2) => {
  const out2 = printed2();
  const print = out2.print;
  // doctest:L17
  const tap = 2;
  const out = 3;
  const printed = 4;
  print("x");
  await tap2.checkText(out2.with(tap + out + printed + t), `x
10`);
});

// A comma expression in a throws example is one expression:
void test("A comma expression in a throws example is one expression:", async (tap2) => {
  // doctest:L29
  await tap2.checkThrows(async () => ((() => { throw new Error("boom"); })(), 1), "Error");
});
