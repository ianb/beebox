// # Names
//
// Setup that declares `t` and `after`: the converted file binds tap's under other names.

import { after as tapAfter, test } from "tap";
import type { ParsedPath } from "node:path";

const t = { label: "mine" };
const after = "later";

tapAfter(async () => {
  void after;
});

// The doctest gave each example 120s before a still-running notice.
void test("line 14", async (tap) => {
  // doctest:L15
  await tap.checkText(t.label, `mine`);
});

// An example import becomes a dynamic import in place, and a type-only import moves to module level:
void test("An example import becomes a dynamic import in place, and a type-only import m...", async (tap) => {
  // doctest:L22
  const { basename } = await import("node:path");

  const parsed: ParsedPath | null = null;
  await tap.checkText(basename("/a/b.txt"), `b.txt`);
});

// A try block's value, which needs the esbuild split:
void test("A try block's value, which needs the esbuild split:", async (tap) => {
  // doctest:L32
  await tap.checkText(
    await (async () => {
    try {
      return JSON.parse("{");
    } catch (e) {
      return e instanceof SyntaxError;
    }
    })(),
    `true`,
  );
});
