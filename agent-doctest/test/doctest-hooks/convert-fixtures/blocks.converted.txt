import { test } from "tap";

const base = 10;

// ## Adding
//
// Adding one to the base:
void test("Adding one to the base:", async (t) => {
  // doctest:L14
  await t.checkText(base + 1, `11`);
});

// A continue block shares the test's variables.
void test("A continue block shares the test's variables.", async (t) => {
  // doctest:L21
  const doubled = base * 2;
  await t.checkText(doubled, `20`);

  // doctest:L27
  await t.checkText(doubled + 1, `21`);
});
