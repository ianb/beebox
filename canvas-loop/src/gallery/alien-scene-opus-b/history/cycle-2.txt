// "Hello, Earth!" — a friendly green alien has just stepped down the ramp of
// its landed saucer onto a dusk meadow, and waves to us. Picture-book
// illustration painted with canvas-loop brushes.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

const W = 800;
const H = 600;

const LINE = "#2a2433";
const GREEN = "#7cc96b";
const GREEN_DARK = "#3f7f4f";
const GREEN_LIGHT = "#b8ea8f";
const WARM = "#ffd98a";
const SHADE = "#4f8a6a";

// ── helpers ────────────────────────────────────────────────────────────
function oval(cx: number, cy: number, rx: number, ry: number, n: number, rot = 0): Vec2[] {
  const pts: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2 + rot;
    pts.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return pts;
}

/** Closed outline of a tube of `width` around a centerline. */
function tube(center: readonly Vec2[], width: number, endWidth = width): Vec2[] {
  const left: Vec2[] = [];
  const right: Vec2[] = [];
  const n = center.length;
  for (let i = 0; i < n; i += 1) {
    const a = center[Math.max(0, i - 1)];
    const b = center[Math.min(n - 1, i + 1)];
    const dx = b[0] - a[0];
    const dy = b[1] - a[1];
    const len = Math.hypot(dx, dy) || 1;
    const nx = -dy / len;
    const ny = dx / len;
    const t = n === 1 ? 0 : i / (n - 1);
    const hw = (width + (endWidth - width) * t) / 2;
    const [x, y] = center[i];
    left.push([x + nx * hw, y + ny * hw]);
    right.push([x - nx * hw, y - ny * hw]);
  }
  return [...left, ...right.reverse()];
}

function ink(s: Sketch, color: string, weight: number, wobble = 0.6): void {
  s.noFill();
  s.stroke(color);
  s.strokeWeight(weight);
  s.brush("ink", { thinning: 0.55 });
  s.gesture({ pressure: "taper", wobble });
}

function closedBelow(top: readonly Vec2[]): Vec2[] {
  return [...top, [W + 20, H + 20], [-20, H + 20]];
}

const BODY: Vec2[] = [
  [294, 348], [274, 392], [262, 445], [266, 492], [296, 508], [364, 508], [394, 492],
  [398, 445], [386, 392], [366, 348],
];
const ARM_WAVE: Vec2[] = [[372, 392], [410, 380], [438, 350], [452, 314]];
const ARM_DOWN: Vec2[] = [[288, 392], [262, 418], [248, 452], [250, 476]];
const LEG_L: Vec2[] = [[314, 500], [312, 524], [310, 546]];
const LEG_R: Vec2[] = [[346, 500], [348, 524], [350, 546]];
const ANT_L: Vec2[] = [[302, 210], [292, 180], [276, 150]];
const ANT_R: Vec2[] = [[360, 208], [372, 176], [392, 146]];

const HILL_FAR: Vec2[] = [
  [-20, 350], [120, 322], [260, 340], [420, 312], [560, 330], [700, 305], [820, 322],
];
const HILL_MID: Vec2[] = [
  [-20, 410], [150, 388], [330, 402], [520, 374], [680, 388], [820, 380],
];

const SAUCER = { x: 610, y: 335 };

// ── layers ─────────────────────────────────────────────────────────────
function sky(s: Sketch): void {
  s.background(
    s.linearGradient(0, 0, 0, 420, [
      [0, "#1c2150"],
      [0.42, "#4a3a7a"],
      [0.72, "#b06a8e"],
      [1, "#f5b67c"],
    ]),
  );
  // soft cloud bands, wash
  s.noFill();
  s.brush("wash", { opacity: 0.35 });
  s.gesture({ pressure: "swell", wobble: 6 });
  s.stroke("#e9a0a6");
  s.strokeWeight(34);
  s.curve([[380, 230], [520, 214], [700, 226], [830, 210]]);
  s.stroke("#7a5a9a");
  s.strokeWeight(26);
  s.curve([[-20, 200], [120, 186], [300, 196]]);
  // moon glow
  s.stroke("#fff1c9");
  s.brush("airbrush", { opacity: 0.22 });
  s.strokeWeight(90);
  s.gesture({ pressure: "even" });
  s.circle(140, 108, 80);
  s.noBrush();
  s.noStroke();
  s.fill("#fbf0cf");
  s.circle(140, 108, 70);
  s.fill("rgba(214,196,160,0.5)");
  s.circle(128, 100, 14);
  s.circle(154, 122, 9);
  // stars
  for (let i = 0; i < 80; i += 1) {
    const x = s.random(W);
    const y = s.random(250) * s.random(0.3, 1);
    const r = s.random(0.8, 2.6);
    s.fill(`rgba(255,248,220,${(0.35 + s.random(0.65)).toFixed(2)})`);
    s.circle(x, y, r);
  }
}

function hills(s: Sketch): void {
  s.noStroke();
  s.fill("#6a5490");
  s.curve(closedBelow(HILL_FAR), true);
  s.clip(s.curvePath(closedBelow(HILL_FAR), true), () => {
    s.noFill();
    s.stroke("#8a6aa2");
    s.brush("wash", { opacity: 0.5 });
    s.strokeWeight(40);
    s.gesture({ pressure: "taper", wobble: 4 });
    s.curve([[-20, 345], [120, 318], [260, 336], [420, 308], [560, 326], [700, 300], [820, 318]]);
  });
  s.noStroke();
  s.fill("#3d5872");
  s.curve(closedBelow(HILL_MID), true);
  s.clip(s.curvePath(closedBelow(HILL_MID), true), () => {
    s.noFill();
    s.stroke("#55708a");
    s.brush("chalk", { opacity: 0.45 });
    s.strokeWeight(14);
    s.gesture({ pressure: "taper", wobble: 3 });
    for (let i = 0; i < 6; i += 1) {
      const y = 400 + i * 10;
      s.curve([[-20 + i * 30, y], [200, y - 14], [420, y - 6], [640, y - 20], [820, y - 10]]);
    }
  });
  // ground
  s.noStroke();
  s.fill("#3e6a45");
  s.curve(closedBelow(GROUND), true);
}

function saucer(s: Sketch): void {
  const { x: cx, y: cy } = SAUCER;
  // beam of warm light spilling down from the hatch: soft wash strokes
  s.group({ opacity: 0.85, blend: "screen" }, () => {
    s.noFill();
    s.stroke("#ffcf7a");
    s.brush("wash", { opacity: 0.35 });
    s.gesture({ pressure: [[0, 0.5], [1, 1]], wobble: 1 });
    for (let i = -3; i <= 3; i += 1) {
      s.strokeWeight(30);
      s.line(cx + i * 6, cy + 16, cx + i * 30, 476);
    }
  });
  // legs
  ink(s, LINE, 3, 0.3);
  s.line(cx - 70, cy + 14, cx - 92, cy + 66);
  s.line(cx + 70, cy + 14, cx + 92, cy + 66);
  s.noBrush();
  s.noStroke();
  // dome
  s.fill(s.radialGradient(cx - 12, cy - 38, 60, [[0, "#f0ffff"], [1, "#8fc9d4"]]));
  s.ellipse(cx, cy - 20, 112, 82);
  // disc
  s.fill(s.linearGradient(cx, cy - 26, cx, cy + 26, [[0, "#c8cddd"], [1, "#6f7590"]]));
  s.ellipse(cx, cy, 224, 54);
  // windows
  for (let i = -2; i <= 2; i += 1) {
    s.fill("#ffe08a");
    s.circle(cx + i * 38, cy + 4 - Math.abs(i) * 2, 12);
  }
  ink(s, LINE, 3);
  s.ellipse(cx, cy, 224, 54);
  s.curve(oval(cx, cy - 20, 56, 41, 10).filter((p) => p[1] < cy - 6));
  s.noBrush();
}

function meadow(s: Sketch): void {
  s.clip(s.curvePath(closedBelow(GROUND), true), () => {
    // light pool from the saucer
    s.noStroke();
    s.fill(s.radialGradient(560, 480, 200, [[0, "rgba(255,214,130,0.45)"], [1, "rgba(255,214,130,0)"]]));
    s.rect(300, 400, 500, 200);
    // painted grass bands
    s.noFill();
    s.brush("bristle", { load: 0.7 });
    s.gesture({ pressure: "press", wobble: 2 });
    const colors = ["#4f8a4c", "#355c3a", "#5c9a52", "#2f4f35"];
    for (let i = 0; i < 14; i += 1) {
      const y = 470 + i * 10;
      s.stroke(colors[i % colors.length]);
      s.strokeWeight(22);
      const dy = s.random(-6, 6);
      s.curve([[-60, y + dy], [200, y - 6], [420, y + 4 - dy], [640, y - 4], [860, y + dy]]);
    }
  });
}

function grassTufts(s: Sketch): void {
  ink(s, "#24402c", 2, 0.3);
  for (let i = 0; i < 70; i += 1) {
    const x = s.random(W);
    const y = s.random(470, 595);
    if (x > 250 && x < 420 && y < 560) continue;
    const h = 8 + (y - 460) * 0.12;
    s.line(x, y, x - 3, y - h);
    s.line(x + 3, y, x + 5, y - h * 0.8);
  }
  // flowers
  s.noBrush();
  s.noStroke();
  for (let i = 0; i < 24; i += 1) {
    const x = s.random(W);
    const y = s.random(480, 595);
    if (x > 250 && x < 420 && y < 560) continue;
    s.fill(s.randomChoice(["#f6d6e0", "#fff3c4", "#c9d8ff"]));
    s.circle(x, y, 4 + (y - 470) * 0.03);
  }
}

/** Fill a shape, shade inside it, then outline it. */
function part(s: Sketch, shape: readonly Vec2[], shade: () => void, outline = 3.5): void {
  s.noBrush();
  s.noStroke();
  s.fill(GREEN);
  s.curve(shape, true);
  s.clip(s.curvePath(shape, true), shade);
  ink(s, LINE, outline);
  s.curve(shape, true);
}

function shadeStroke(s: Sketch, weight: number, pts: readonly Vec2[], opacity = 0.75): void {
  s.noFill();
  s.stroke(SHADE);
  s.brush("wash", { opacity, blend: "multiply" });
  s.strokeWeight(weight);
  s.gesture({ pressure: "swell", wobble: 2 });
  s.curve(pts);
}

function rimStroke(s: Sketch, weight: number, pts: readonly Vec2[], opacity = 0.7): void {
  s.noFill();
  s.stroke(WARM);
  s.brush("wash", { opacity, blend: "screen" });
  s.strokeWeight(weight);
  s.gesture({ pressure: "swell", wobble: 1 });
  s.curve(pts);
}

function alien(s: Sketch): void {
  // ground shadow
  s.noFill();
  s.stroke("#1b2a20");
  s.brush("wash", { opacity: 0.55, blend: "multiply" });
  s.strokeWeight(30);
  s.gesture({ pressure: "swell", wobble: 2 });
  s.line(250, 552, 420, 552);

  const legL = tube(LEG_L, 24);
  const legR = tube(LEG_R, 24);
  for (const leg of [legL, legR]) {
    part(s, leg, () => {
      shadeStroke(s, 16, [[leg[0][0] - 14, 500], [leg[0][0] - 12, 550]]);
    });
  }
  // feet
  for (const fx of [304, 356]) {
    const foot = oval(fx, 548, 24, 10, 10);
    part(s, foot, () => shadeStroke(s, 12, [[fx - 26, 556], [fx + 26, 556]]));
  }
  // lowered arm (behind body)
  const armDown = tube(ARM_DOWN, 26, 20);
  part(s, armDown, () => shadeStroke(s, 18, [[262, 392], [236, 440], [238, 480]]));
  part(s, oval(250, 482, 16, 15, 10), () => shadeStroke(s, 14, [[236, 486], [256, 500]]));
  // waving arm (behind body)
  const armWave = tube(ARM_WAVE, 26, 20);
  part(s, armWave, () => {
    shadeStroke(s, 16, [[372, 404], [420, 392], [452, 350], [466, 316]]);
    rimStroke(s, 8, [[380, 382], [420, 368], [440, 336]], 0.6);
  });

  // body
  part(s, BODY, () => {
    shadeStroke(s, 60, [[256, 360], [262, 440], [296, 516]], 0.8);
    shadeStroke(s, 40, [[280, 350], [330, 366], [380, 350]], 0.7);
    shadeStroke(s, 30, [[270, 512], [330, 520], [400, 506]], 0.6);
    rimStroke(s, 16, [[398, 384], [406, 440], [398, 500]]);
    // belly patch: soft wash, darker centre
    s.stroke(SHADE);
    s.brush("wash", { opacity: 0.45, blend: "multiply" });
    s.strokeWeight(40);
    s.gesture({ pressure: "swell", wobble: 2 });
    s.line(334, 424, 334, 472);
  }, 4);

  // head over the body
  part(s, HEAD, () => {
    shadeStroke(s, 70, [[220, 240], [242, 330], [330, 368], [432, 330]], 0.75);
    s.noFill();
    s.stroke(GREEN_LIGHT);
    s.brush("airbrush", { opacity: 0.7 });
    s.strokeWeight(60);
    s.gesture({ pressure: "swell" });
    s.curve([[280, 226], [340, 216], [392, 236]]);
    rimStroke(s, 22, [[412, 226], [432, 270], [420, 318]], 0.8);
  }, 4);

  // hand that waves, with three little fingers
  const HX = 456;
  const HY = 302;
  for (const [dx, dy, a] of [[-12, -18, -0.5], [0, -22, 0], [12, -17, 0.5]] as const) {
    part(s, oval(HX + dx, HY + dy, 6, 10, 8, a), () => undefined, 2.5);
  }
  part(s, oval(HX, HY, 18, 16, 10), () => {
    shadeStroke(s, 14, [[440, 314], [460, 320], [472, 306]]);
    rimStroke(s, 6, [[466, 290], [474, 302]], 0.7);
  });

  // antennae
  ink(s, LINE, 4, 0.4);
  s.curve(ANT_L);
  s.curve(ANT_R);
  for (const [x, y] of [[274, 146], [394, 142]] as const) {
    s.noFill();
    s.stroke("#ffe27a");
    s.brush("airbrush", { opacity: 0.45 });
    s.strokeWeight(40);
    s.gesture({ pressure: "even" });
    s.line(x, y, x + 1, y);
    s.noBrush();
    s.noStroke();
    s.fill("#ffd84a");
    s.circle(x, y, 20);
    s.fill("#fff6c0");
    s.circle(x - 3, y - 3, 6);
    ink(s, LINE, 3);
    s.circle(x, y, 20);
  }

  // face
  s.noBrush();
  s.noStroke();
  s.fill("#ffffff");
  s.ellipse(290, 276, 54, 62);
  s.ellipse(372, 276, 54, 62);
  s.fill(LINE);
  s.circle(294, 282, 26);
  s.circle(368, 282, 26);
  s.fill("#ffffff");
  s.circle(299, 276, 8);
  s.circle(373, 276, 8);
  ink(s, LINE, 3, 0.2);
  s.ellipse(290, 276, 54, 62);
  s.ellipse(372, 276, 54, 62);
  // cheeks
  s.noBrush();
  s.noStroke();
  s.fill("rgba(240,130,130,0.55)");
  s.ellipse(254, 314, 26, 14);
  s.ellipse(408, 314, 26, 14);
  // smile
  ink(s, LINE, 4, 0);
  s.curve([[310, 318], [320, 328], [331, 331], [342, 328], [352, 318]]);

  // wave motion lines
  ink(s, "#fff3d0", 2.5, 0.4);
  s.curve([[484, 278], [494, 298], [490, 320]]);
  s.curve([[498, 268], [512, 296], [506, 328]]);
  s.noBrush();
}

function fireflies(s: Sketch): void {
  const flies: Vec2[] = [[180, 420], [482, 430], [530, 250], [90, 505], [700, 525], [210, 330], [745, 430]];
  for (const [x, y] of flies) {
    s.noFill();
    s.stroke("#fff3a0");
    s.brush("airbrush", { opacity: 0.55, blend: "screen" });
    s.strokeWeight(34);
    s.gesture({ pressure: "even" });
    s.line(x, y, x + 1, y);
    s.noBrush();
    s.noStroke();
    s.fill("#fffbe0");
    s.circle(x, y, 5);
  }
}

export function setup(s: Sketch): void {
  s.createCanvas(W, H);
}

export function draw(s: Sketch): void {
  sky(s);
  hills(s);
  saucer(s);
  meadow(s);
  grassTufts(s);
  alien(s);
  fireflies(s);
}
