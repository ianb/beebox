// Acorn Lane: a busy picture-book street. Every townsperson idles; click one
// and it performs its little act, sometimes setting off a neighbour.
import type { DeepReadonly, Msg, ParamValues, ParamsDecl, Util, View } from "@ianbicking/canvas-loop";
import type { Id } from "./things-tea.js";
import { CHAIN, H, NOT_STARTED, W, act, hit } from "./things-tea.js";
import { paintLights, paintTown } from "./town-tea.js";
import * as A from "./actors-tea.js";

export const params = {
  time: { type: "select", options: ["morning", "dusk", "night"] as const, default: "morning" },
  rain: { type: "boolean", default: false },
  reset: { type: "trigger" },
} as const satisfies ParamsDecl;

export const canvas = { width: W, height: H };

interface Cue {
  readonly id: Id;
  readonly at: number;
}

export interface Model {
  readonly frame: number;
  readonly hover: Id | "";
  readonly start: Readonly<Record<Id, number>>;
  readonly queue: readonly Cue[];
  readonly bloom: number;
}

function fresh(frame: number): Model {
  return { frame, hover: "", start: NOT_STARTED, queue: [], bloom: 0 };
}

export function init(): Model {
  return fresh(0);
}

/** Start an act on `id` at frame `at` unless it is already playing; queue its chain reactions. */
function begin(model: DeepReadonly<Model>, id: Id, at: number): Model {
  if (act({ frame: at, start: model.start }, id) !== null) return model;
  const chained = (CHAIN[id] ?? []).map((c) => ({ id: c.id, at: at + c.delay }));
  return {
    ...model,
    start: { ...model.start, [id]: at },
    queue: [...model.queue, ...chained],
    bloom: id === "window" ? Math.min(3, model.bloom + 1) : model.bloom,
  };
}

function tick(model: DeepReadonly<Model>, frame: number): Model {
  const due = model.queue.filter((c) => c.at <= frame);
  const rest = model.queue.filter((c) => c.at > frame);
  return due.reduce<Model>((m, c) => begin(m, c.id, frame), { ...model, frame, queue: rest });
}

export function update(model: DeepReadonly<Model>, msg: Msg, u: Util<typeof params>): Model {
  switch (msg.type) {
    case "tick":
      return tick(model, msg.frame);
    case "mousemove": {
      const id = hit(model, msg.x, msg.y) ?? "";
      return id === model.hover ? model : { ...model, hover: id };
    }
    case "mousedown": {
      const id = hit(model, msg.x, msg.y);
      if (id === null) return model;
      const at = model.frame + 1;
      if (act({ frame: at, start: model.start }, id) !== null) {
        u.handled(`${id} (busy)`);
        return model;
      }
      u.handled(id);
      return begin(model, id, at);
    }
    case "mouseup":
    case "keydown":
    case "keyup":
    case "param":
      return model;
    case "trigger":
      if (msg.name !== "reset") return model;
      u.handled("reset");
      return fresh(model.frame);
  }
}

const SHADE: Readonly<Record<ParamValues<typeof params>["time"], string>> = {
  morning: "",
  dusk: "#f2b48c",
  night: "#5a64a8",
};

export function draw(v: View, model: DeepReadonly<Model>, p: ParamValues<typeof params>): void {
  const night = p.time === "night";
  const s: A.Scene = { ...model, night };
  v.cache("town", () => paintTown(v));
  // back to front
  A.chimney(v, s);
  A.flag(v, s);
  A.clockHands(v, s);
  A.windowAndGoat(v, s);
  A.owl(v, s);
  A.cat(v, s);
  A.painter(v, s);
  A.baker(v, s);
  A.mailCarrier(v, s);
  A.mailbox(v, s);
  A.hydrant(v, s);
  A.lamp(v, s);
  A.bucket(v, s);
  A.dog(v, s);
  A.grocer(v, s);
  A.duck(v, s);
  A.pigeons(v, s);
  A.snail(v, s);
  A.manhole(v, s);
  A.bus(v, s);
  if (p.rain) {
    v.noStroke();
    v.fill("rgba(90,100,130,0.25)");
    v.rect(0, 0, W, H);
  }
  const shade = SHADE[p.time];
  if (shade !== "") {
    v.blendMode("multiply");
    v.noStroke();
    v.fill(shade);
    v.rect(0, 0, W, H);
    v.blendMode("screen");
    v.cache(`lights-${p.time}`, () => paintLights(v, night));
    v.blendMode("normal");
  }
  if (p.rain) A.rain(v, model.frame);
  A.bubbles(v, s);
  A.hoverGlint(v, s);
}

export const W = 1000;

export const BUS_HOME = 590;
export const BUS_Y = 582;
export const BUS_W = 220;
export const BUS_H = 112;

/** Act progress 0..1 while an act runs, otherwise null. */
export function act(c: Clock, id: Id): number | null {
  const t = (c.frame - c.start[id]) / DUR[id];
  return t >= 0 && t < 1 ? t : null;
}

/** Bus left edge: drives off right, wraps round the block, and creeps back to the stop. */
export function busX(c: Clock): number {
  const t = act(c, "bus");
  if (t === null) return BUS_HOME;
  if (t < 0.45) {
    const k = t / 0.45;
    return BUS_HOME + k * k * (W - BUS_HOME + 40);
  }
  if (t < 0.55) return W + 100; // round the corner, out of sight
  const k = (t - 0.55) / 0.45;
  const ease = 1 - (1 - k) * (1 - k);
  return -BUS_W - 60 + ease * (BUS_HOME + BUS_W + 60);
}

/** Snail left edge: it crawls the curb and loops once every ~80 s. */
export function snailX(frame: number): number {
  return 40 + ((frame * 0.05) % 300);
}

export function boxOf(c: Clock, id: Id): Box {
  if (id === "bus") return { x: busX(c), y: BUS_Y, w: BUS_W, h: BUS_H };
  if (id === "snail") return { x: snailX(c.frame), y: 578, w: 36, h: 26 };
  return STATIC[id];
}

/** Front-most first: the order a click is offered to things. */
const HIT_ORDER: readonly Id[] = [
  "bus",
  "snail",
  "manhole",
  "duck",
  "dog",
  "bucket",
  "hydrant",
  "mailbox",
  "baker",
  "mail",
  "grocer",
  "lamp",
  "cat",
  "painter",
  "owl",
  "window",
  "chimney",
  "pigeons",
  "clock",
  "flag",
];

export function hit(c: Clock, x: number, y: number): Id | null {
  for (const id of HIT_ORDER) {
    const b = boxOf(c, id);
    if (x >= b.x && x <= b.x + b.w && y >= b.y && y <= b.y + b.h) return id;
  }
  return null;
}

===== actors-tea.ts =====
// Every moving thing, drawn each frame from sprites and a few cheap marks.
import type { Sprite, View } from "@ianbicking/canvas-loop";
import type { Clock, Id } from "./things-tea.js";
import { BUS_Y, FEET, IDS, W, act, boxOf, busX, snailX } from "./things-tea.js";
import * as S from "./folk-tea.js";

export interface Scene extends Clock {
  readonly hover: Id | "";
  readonly bloom: number;
  readonly night: boolean;
}

interface Place {
  readonly x: number;
  readonly y: number;
  readonly px: number;
  readonly py: number;
  readonly rot?: number;
  readonly flip?: boolean;
  readonly alpha?: number;
  readonly scale?: number;
}

/** Draw a sprite with its pivot (px, py) at (x, y), turned, flipped and scaled about the pivot. */
export function put(v: View, s: Sprite, at: Place): void {
  v.push();
  v.translate(at.x, at.y);
  if (at.rot !== undefined) v.rotate(at.rot);
  if (at.flip === true) v.scale(-1, 1);
  if (at.scale !== undefined) v.scale(at.scale);
  v.image(s, -at.px, -at.py, at.alpha ?? 1);
  v.pop();
}

/** Hover cue: a little hop while the pointer is over an idle thing. */
export function hop(s: Scene, id: Id): number {
  return s.hover === id && act(s, id) === null ? -Math.abs(Math.sin(s.frame * 0.22)) * 6 : 0;
}

/** Hover cue for fixed things: a small wiggle. */
function wiggle(s: Scene, id: Id): number {
  return s.hover === id && act(s, id) === null ? Math.sin(s.frame * 0.5) * 0.06 : 0;
}

function puff(v: View, x: number, y: number, r: number, alpha: number): void {
  v.noStroke();
  v.fill(`rgba(250,250,250,${alpha.toFixed(3)})`);
  v.circle(x, y, r * 2);
}

function drops(v: View, color: string, pts: readonly (readonly [number, number])[]): void {
  v.noStroke();
  v.fill(color);
  for (const [x, y] of pts) v.ellipse(x, y, 4, 6);
}

const bell = (t: number, a: number, b: number): number => (t < a || t > b ? 0 : Math.sin(((t - a) / (b - a)) * Math.PI));

// ── Bakery row ─────────────────────────────────────────────────────

export function chimney(v: View, s: Scene): void {
  const t = act(s, "chimney");
  const shake = t === null ? wiggle(s, "chimney") : Math.sin(t * 60) * 0.08 * (1 - t);
  put(v, S.chimney(v), { x: 185, y: 252, px: 17, py: 56, rot: shake });
  for (let i = 0; i < 6; i += 1) {
    const age = ((s.frame + i * 20) % 120) / 120;
    puff(v, 186 + Math.sin(age * 5 + i) * 8 + age * 30, 196 - age * 90, 5 + age * 12, (1 - age) * 0.55);
  }
  if (t !== null) {
    v.noFill();
    v.noBrush();
    for (let i = 0; i < 3; i += 1) {
      const k = t * 1.6 - i * 0.25;
      if (k <= 0 || k >= 1) continue;
      v.stroke(`rgba(90,80,90,${(0.8 * (1 - k)).toFixed(3)})`);
      v.strokeWeight(5 * (1 - k) + 1);
      v.ellipse(186 + k * 20, 190 - k * 150, 18 + k * 50, 8 + k * 20);
    }
  }
}

export function windowAndGoat(v: View, s: Scene): void {
  const t = act(s, "window");
  const open = t !== null && t > 0.04 && t < 0.94;
  const x = 50;
  const y = 282 + hop(s, "window");
  if (open) {
    const rise = Math.min(1, (t - 0.04) / 0.12) * Math.min(1, (0.94 - t) / 0.1);
    put(v, S.windowSprite(v, true), { x, y, px: 0, py: 0 });
    const gy = y + 120 - rise * 52;
    v.clip([[x + 16, y + 6], [x + 64, y + 6], [x + 64, y + 66], [x + 16, y + 66]], () => {
      put(v, S.folk(v, "goat-water"), { x: x + 40, y: gy, px: 32, py: 96 });
    });
    v.clip([[x, y + 64], [x + 80, y + 64], [x + 80, y + 86], [x, y + 86]], () => put(v, S.windowSprite(v, true), { x, y, px: 0, py: 0 }));
    const pour = t > 0.18 && t < 0.8;
    const tilt = pour ? 0.55 + Math.sin(s.frame * 0.2) * 0.08 : 0;
    // the can pokes out over the sill, held at the goat's hands
    if (rise > 0.6) put(v, S.wateringCan(v), { x: x + 66, y: gy - 38, px: 6, py: 10, rot: tilt });
    if (pour) {
      const sx = x + 66 + Math.cos(tilt) * 23 - Math.sin(tilt) * -8;
      const sy = gy - 38 + Math.sin(tilt) * 23 + Math.cos(tilt) * -8;
      const pts: [number, number][] = [];
      for (let i = 0; i < 5; i += 1) {
        const k = ((s.frame * 2 + i * 7) % 35) / 35;
        pts.push([sx - k * 22, sy + k * k * (y + 56 - sy)]);
      }
      // what the box cannot hold drips out underneath, onto whoever sits below
      for (let i = 0; i < 4; i += 1) {
        const k = ((s.frame * 1.5 + i * 13) % 52) / 52;
        if (t > 0.3) pts.push([x + 28 + (i % 2) * 8, y + 82 + k * 40]);
      }
      drops(v, "rgba(110,170,230,0.85)", pts);
    }
  } else {
    put(v, S.windowSprite(v, false), { x, y, px: 0, py: 0 });
  }
  // flowers in the box: they sway, and grow while watered
  const grow = t !== null && t > 0.18 ? Math.min(1, (t - 0.18) / 0.5) : 0;
  const tall = 10 + s.bloom * 6 + grow * 6;
  const colors = ["#e84a6a", "#f2c84b", "#b46ad8", "#ff8a3a", "#e84a6a"];
  for (let i = 0; i < 5; i += 1) {
    const fx = x + 16 + i * 12;
    const sway = Math.sin(s.frame * 0.05 + i) * 2;
    v.noBrush();
    v.noFill();
    v.stroke("#3f7a3a");
    v.strokeWeight(2);
    v.line(fx, y + 67, fx + sway, y + 67 - tall);
    v.noStroke();
    v.fill(colors[i] ?? "#e84a6a");
    v.circle(fx + sway, y + 67 - tall, 9 + grow * 3 + s.bloom);
    v.fill("#fff3b0");
    v.circle(fx + sway, y + 67 - tall, 3);
  }
}

export function cat(v: View, s: Scene): void {
  const t = act(s, "cat");
  const leap = t === null ? 0 : Math.sin(Math.min(1, t / 0.7) * Math.PI) * 46;
  const mode = t !== null && t < 0.75 ? "arch" : "sit";
  const x = 52;
  const y = 360 - leap + hop(s, "cat");
  // tail first, behind
  const flick = t === null ? Math.sin(s.frame * 0.09) * 10 + Math.sin(s.frame * 0.23) * 4 : 0;
  v.noBrush();
  v.noFill();
  v.stroke("#8e8a9a");
  v.strokeWeight(6);
  if (mode === "sit") {
    v.curve([[x + 36, y + 50], [x + 50, y + 56], [x + 56 + flick * 0.3, y + 64], [x + 54 + flick, y + 78]]);
  } else {
    v.curve([[x + 42, y + 34], [x + 50, y + 20], [x + 52, y + 4]]);
  }
  put(v, S.cat(v, mode), { x, y, px: 0, py: 0 });
}

export function baker(v: View, s: Scene): void {
  const t = act(s, "baker");
  const x = 114;
  const pose: S.Pose = t === null || t > 0.82 ? "baker-idle" : t < 0.28 ? "baker-reach" : "baker-up";
  const bob = Math.sin(s.frame * 0.07) * 1.2 + hop(s, "baker");
  put(v, S.folk(v, pose), { x, y: FEET + bob, px: 32, py: 96 });
  if (t === null || t > 0.82) return;
  let lx: number;
  let ly: number;
  if (t < 0.28) {
    const k = t / 0.28;
    lx = 200 - k * 58;
    ly = 520 - k * 12;
    v.noBrush();
    v.stroke("#8a5a3a");
    v.strokeWeight(3);
    v.line(lx + 26, ly + 2, 142, 543);
  } else {
    lx = x;
    ly = FEET - 92 + bob;
  }
  put(v, S.loaf(v), { x: lx, y: ly, px: 22, py: 14 });
  for (let i = 0; i < 6; i += 1) {
    const age = ((s.frame * 1.3 + i * 11) % 66) / 66;
    puff(v, lx + Math.sin(age * 6 + i * 2) * 7, ly - 8 - age * 46, 3 + age * 8, (1 - age) * 0.7);
  }
}

export function painter(v: View, s: Scene): void {
  const t = act(s, "painter");
  const wob = t === null ? wiggle(s, "painter") : Math.sin(t * Math.PI * 9) * 0.2 * (1 - t) * Math.min(1, t * 8);
  const hang = t !== null && t < 0.8;
  put(v, S.ladder(v, hang), { x: 536, y: 568, px: 16, py: 236, rot: wob });
  if (!hang) {
    // idle: the brush arm dabs, leaving a dab of yellow on the wall
    const dab = Math.sin(s.frame * 0.12);
    v.noStroke();
    v.fill("rgba(247,214,92,0.9)");
    v.ellipse(592 + dab * 2, 382 + dab * 7, 7, 10);
  }
  if (t !== null && t < 0.9) {
    // paint drips falling to the bucket

/** Some lines depend on what the neighbours are doing. */
function say(s: Scene, id: Id, line: string): string {
  if (id === "duck" && act(s, "bus") !== null) return "QUACK! You're late!";
  if (id === "dog" && act(s, "baker") !== null) return "Sniff sniff... bread?";
  if (id === "dog" && act(s, "hydrant") !== null) return "Brrr! Wet dog!";
  if (id === "owl" && s.night) return "Hoo! It's MY bedtime!";
  if (id === "cat" && act(s, "window") !== null) return "MRRAOW! Wet!";
  return line;
}

interface Placed {
  readonly x: number;
  readonly y: number;
  readonly w: number;
}

const overlaps = (a: Placed, b: Placed): boolean => Math.abs(a.x - b.x) < (a.w + b.w) / 2 + 4 && Math.abs(a.y - b.y) < 29;

export function bubbles(v: View, s: Scene): void {
  const placed: Placed[] = [];
  for (const id of IDS) {
    const b = BUBBLES[id];
    const t = act(s, id);
    if (b === undefined || t === null || t < b.from || t > b.to) continue;
    const box = boxOf(s, id);
    const text = say(s, id, b.text);
    const w = text.length * 7.2 + 18;
    const cx = Math.min(W - w / 2 - 6, Math.max(w / 2 + 6, box.x + box.w / 2 + b.dx));
    const tipX = cx - b.dx * 0.15;
    const tipY = Math.max(24, box.y + b.dy - 14) + 24;
    // nudge up past any bubble already placed, so lines never cover each other
    let cy = tipY - 24;
    for (let tries = 0; tries < 5 && placed.some((p) => overlaps(p, { x: cx, y: cy, w })); tries += 1) cy -= 30;
    placed.push({ x: cx, y: cy, w });
    v.noBrush();
    v.fill("#fffdf4");
    v.stroke("#3b2a24");
    v.strokeWeight(2);
    v.triangle(cx - 6, cy + 10, cx + 6, cy + 10, tipX, tipY);
    v.rect(cx - w / 2, cy - 13, w, 25);
    v.noStroke();
    v.rect(cx - 7, cy + 8, 14, 5);
    v.fill("#3b2a24");
    v.textSize(13);
    v.textAlign("center", "middle");
    v.text(text, cx, cy);
  }
}

===== paint-tea.ts =====
// Shared brush helpers and the townsfolk painter. Everything here paints in
// sprite-local coordinates; nothing here keeps state.
import type { Vec2, View } from "@ianbicking/canvas-loop";

export const INK = "#3b2a24";

/** A deterministic 0..1 hash of an integer, for scattering marks in draw (which has no randomness). */
export function hash(n: number): number {
  const s = Math.sin(n * 127.1 + 311.7) * 43_758.5453;
  return s - Math.floor(s);
}

/** Ink outline state: brushed ink, even pressure, a little hand wobble. */
export function ink(v: View, weight = 2.2, seed?: number): void {
  v.brush("ink", { thinning: 0.35 });
  v.stroke(INK);
  v.strokeWeight(weight);
  v.gesture(seed === undefined ? { pressure: "even", wobble: 0.7 } : { pressure: "even", wobble: 0.7, seed });
}

/** Fill with a flat colour and outline with ink, then draw. */
export function inked(v: View, fill: string, body: () => void, weight = 2.2): void {
  v.fill(fill);
  ink(v, weight);
  body();
}

/** A flat fill with no outline. */
export function flat(v: View, fill: string, body: () => void): void {
  v.noStroke();
  v.noBrush();
  v.fill(fill);
  body();
}

/** A soft wash shadow inside a closed outline. */
export function shadeInside(v: View, pts: readonly Vec2[], shade: string, shift: Vec2 = [6, 6]): void {
  v.clip(v.curvePath(pts, true), () => {
    v.noStroke();
    v.noBrush();
    v.fill(shade);
    v.curve(
      v.offset(pts, 2).map(([x, y]): Vec2 => [x + shift[0], y + shift[1]]),
      true,
    );
  });
}

/** An ellipse as points, for curve/clip/offset. */
export function ellipsePts(cx: number, cy: number, rx: number, ry: number, n = 10): Vec2[] {
  const out: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    out.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return out;
}

/** Plain text label in ink. */
export function label(v: View, text: string, x: number, y: number, size = 12, color = INK): void {
  v.noStroke();
  v.noBrush();
  v.fill(color);
  v.textSize(size);
  v.textAlign("center", "middle");
  v.text(text, x, y);
}

// ── Townsfolk (standing, 64×100 sprite, feet at (32, 96)) ─────────────

export type Species = "bear" | "rabbit" | "fox" | "pig" | "goat" | "duck" | "hippo";

export interface Arms {
  readonly l: Vec2;
  readonly r: Vec2;
}

export interface Folk {
  readonly species: Species;
  readonly fur: string;
  readonly muzzle: string;
  readonly shirt: string;
  readonly trim: string;
  readonly hat: "chef" | "cap" | "beret" | "none";
  readonly arms: Arms;
  /** Eyes: open dots, or happy closed arcs. */
  readonly eyes?: "open" | "happy" | "wide";
}

export const FOLK_W = 64;
export const FOLK_H = 100;
const HEAD: Vec2 = [32, 42];
const HEAD_R = 15;
const SHOULDER_L: Vec2 = [21, 62];
const SHOULDER_R: Vec2 = [43, 62];

const BODY: readonly Vec2[] = [
  [32, 55],
  [44, 60],
  [47, 76],
  [44, 90],
  [32, 92],
  [20, 90],
  [17, 76],
  [20, 60],
];

function arm(v: View, f: Folk, from: Vec2, to: Vec2): void {
  const mid: Vec2 = [(from[0] + to[0]) / 2 + (to[0] < from[0] ? -2 : 2), (from[1] + to[1]) / 2 + 2];
  inked(v, f.shirt, () => v.path(v.tube([from, mid, to], { width: 8, profile: [[0, 1], [1, 0.8]] })), 1.8);
  inked(v, f.fur, () => v.circle(to[0], to[1], 8), 1.6);
}

function ears(v: View, f: Folk): void {
  const [hx, hy] = HEAD;
  switch (f.species) {
    case "bear":
    case "hippo": {
      const r = f.species === "bear" ? 6 : 4;
      for (const dx of [-11, 11]) {
        inked(v, f.fur, () => v.circle(hx + dx, hy - 12, r * 2));
        flat(v, f.muzzle, () => v.circle(hx + dx, hy - 12, r));
      }
      return;
    }
    case "rabbit": {
      for (const dx of [-6, 6]) {
        const pts = ellipsePts(hx + dx, hy - 25, 5, 15);
        inked(v, f.fur, () => v.curve(pts, true));
        flat(v, "#f2a7b8", () => v.ellipse(hx + dx, hy - 24, 4, 20));
      }
      return;
    }
    case "fox": {
      for (const s of [-1, 1]) {
        const pts: Vec2[] = [
          [hx + s * 5, hy - 12],
          [hx + s * 15, hy - 27],
          [hx + s * 15, hy - 6],
        ];
        inked(v, f.fur, () => v.polygon(pts));
        flat(v, INK, () => v.polygon([[hx + s * 12, hy - 18], [hx + s * 15, hy - 26], [hx + s * 15, hy - 16]]));
      }
      return;
    }
    case "pig": {
      for (const s of [-1, 1]) {
        inked(v, f.fur, () =>
          v.polygon([
            [hx + s * 5, hy - 13],
            [hx + s * 17, hy - 18],
            [hx + s * 13, hy - 5],
          ]),
        );
      }
      return;
    }
    case "goat": {
      for (const s of [-1, 1]) {
        v.noFill();
        ink(v, 3.4);
        v.stroke("#8a7a60");
        v.curve([
          [hx + s * 6, hy - 11],
          [hx + s * 12, hy - 18],
          [hx + s * 18, hy - 15],
        ]);
        inked(v, f.fur, () => v.curve(ellipsePts(hx + s * 16, hy - 6, 7, 3.5, 8), true), 1.6);
      }
      return;
    }
    case "duck":
      inked(v, f.fur, () => v.curve([[hx - 2, hy - 13], [hx + 2, hy - 22], [hx + 5, hy - 14]], false), 1.6);
      return;
  }
}

function face(v: View, f: Folk): void {
  const [hx, hy] = HEAD;
  if (f.species === "duck") {
    inked(v, "#f29a2e", () => v.curve(ellipsePts(hx, hy + 6, 10, 4.5, 10), true), 1.8);
    v.noFill();
    ink(v, 1.2);
    v.line(hx - 8, hy + 6, hx + 8, hy + 6);
  } else if (f.species === "pig") {
    inked(v, "#f0a2a8", () => v.ellipse(hx, hy + 5, 13, 9), 1.8);
    flat(v, INK, () => {
      v.ellipse(hx - 2.5, hy + 5, 2.2, 3);
      v.ellipse(hx + 2.5, hy + 5, 2.2, 3);
    });
  } else {
    const long = f.species === "fox" || f.species === "goat";
    flat(v, f.muzzle, () => v.ellipse(hx, hy + 6, long ? 14 : 16, long ? 12 : 11));
    flat(v, INK, () => v.ellipse(hx, hy + 2.5, 5, 3.5));
    v.noFill();
    ink(v, 1.3);
    v.curve([[hx - 3.5, hy + 7], [hx, hy + 8.5], [hx + 3.5, hy + 7]]);
  }
  if (f.species === "goat") {
    inked(v, f.muzzle, () => v.polygon([[hx - 3, hy + 11], [hx + 3, hy + 11], [hx, hy + 20]]), 1.4);
  }
  const eyes = f.eyes ?? "open";
  if (eyes === "happy") {
    v.noFill();
    ink(v, 1.6);
    v.curve([[hx - 8, hy - 2], [hx - 5.5, hy - 4.5], [hx - 3, hy - 2]]);
    v.curve([[hx + 3, hy - 2], [hx + 5.5, hy - 4.5], [hx + 8, hy - 2]]);
  } else {
    const r = eyes === "wide" ? 4.5 : 3;
    if (eyes === "wide") flat(v, "#ffffff", () => {
      v.circle(hx - 5.5, hy - 3, 9);
      v.circle(hx + 5.5, hy - 3, 9);
    });
    flat(v, INK, () => {
      v.ellipse(hx - 5.5, hy - 3, r, r * 1.3);
      v.ellipse(hx + 5.5, hy - 3, r, r * 1.3);
    });
    flat(v, "#ffffff", () => {
      v.circle(hx - 5, hy - 4, 1.4);
      v.circle(hx + 6, hy - 4, 1.4);
    });
  }
  flat(v, "rgba(240,120,120,0.45)", () => {
    v.circle(hx - 10, hy + 4, 5);
    v.circle(hx + 10, hy + 4, 5);
  });
}

function hat(v: View, f: Folk): void {
  const [hx, hy] = HEAD;
  switch (f.hat) {
    case "chef":
      inked(v, "#ffffff", () => {
        v.curve([[hx - 11, hy - 12], [hx - 15, hy - 24], [hx - 6, hy - 32], [hx + 6, hy - 32], [hx + 15, hy - 24], [hx + 11, hy - 12]], true);
      }, 1.8);
      inked(v, "#f3eee4", () => v.rect(hx - 11, hy - 15, 22, 5), 1.6);
      return;
    case "cap":
      inked(v, f.trim, () => v.path([["move", hx - 13, hy - 9], ["quad", hx, hy - 26, hx + 13, hy - 9], ["close"]]), 1.8);
      inked(v, f.trim, () => v.rect(hx + 4, hy - 11, 15, 4), 1.6);
      return;
    case "beret":
      inked(v, f.trim, () => v.curve(ellipsePts(hx - 2, hy - 13, 15, 6, 10), true), 1.8);
      flat(v, f.trim, () => v.rect(hx - 3, hy - 22, 3, 5));
      return;
    case "none":
      return;
  }
}

/** Paint a standing townsperson into a 64×100 sprite. */
export function paintFolk(v: View, f: Folk): void {
  // legs and feet
  flat(v, "rgba(60,40,30,0.18)", () => v.ellipse(32, 96, 40, 7));
  inked(v, f.fur, () => {
    v.rect(23, 86, 7, 8);
    v.rect(34, 86, 7, 8);
  }, 1.6);
  const footColor = f.species === "duck" ? "#f29a2e" : INK;
  flat(v, footColor, () => {
    v.ellipse(25, 95, 12, 5);
    v.ellipse(39, 95, 12, 5);
  });
  // body
  inked(v, f.shirt, () => v.curve(BODY, true));
  shadeInside(v, BODY, "rgba(40,20,40,0.18)", [7, 5]);
  v.clip(v.curvePath(BODY, true), () => {
    v.noFill();
    v.stroke("rgba(40,20,40,0.6)");
    v.brush("pencil", { opacity: 0.8 });
    v.strokeWeight(1.6);
    v.gesture({ pressure: "taper", wobble: 0.4 });
    for (let i = 0; i < 7; i += 1) v.line(30 + i * 4, 92, 46 + i * 4, 60);
  });
  flat(v, f.trim, () => v.rect(18, 80, 28, 4));
  if (f.hat === "chef") {
    // apron
    inked(v, "#ffffff", () => v.curve([[24, 66], [40, 66], [42, 88], [32, 90], [22, 88]], true), 1.6);
  }
  arm(v, f, SHOULDER_L, f.arms.l);
  arm(v, f, SHOULDER_R, f.arms.r);
  ears(v, f);
  const head = ellipsePts(HEAD[0], HEAD[1], HEAD_R, HEAD_R - 1, 12);
  inked(v, f.fur, () => v.curve(head, true));
  shadeInside(v, head, "rgba(60,30,20,0.16)", [5, 5]);
  face(v, f);
  hat(v, f);
}

// The still town: sky, buildings, sidewalk, road. Painted once into a cache.
import type { Vec2, View } from "@ianbicking/canvas-loop";
import { INK, ellipsePts, flat, hash, ink, inked, label } from "./paint-tea.js";
import { CURB, H, ROAD_END, STREET, W } from "./things-tea.js";

function sky(v: View): void {
  v.noStroke();
  v.fill(v.linearGradient(0, 0, 0, STREET, [
    [0, "#8ec9ec"],
    [0.6, "#cfe9f4"],
    [1, "#f3f0dc"],
  ]));
  v.rect(0, 0, W, STREET);
  // clouds: soft wash puffs
  v.noFill();
  v.stroke("rgba(255,255,255,0.85)");
  v.brush("wash", { opacity: 0.8 });
  v.gesture({ pressure: "swell", wobble: 3 });
  for (const [x, y, w] of [[120, 70, 140], [430, 50, 110], [690, 90, 160], [300, 140, 80]] as const) {
    v.strokeWeight(34);
    v.line(x, y, x + w, y + 4);
    v.strokeWeight(26);
    v.line(x + w * 0.2, y - 16, x + w * 0.7, y - 14);
  }
  // distant hills and rooftops
  flat(v, "#a9cf9a", () => v.curve([[-20, 330], [150, 270], [330, 300], [520, 250], [760, 290], [1020, 240], [1020, 560], [-20, 560]], true));
  flat(v, "#b9b3d4", () => {
    for (let i = 0; i < 14; i += 1) {
      const x = i * 75 + hash(i) * 30;
      const h = 60 + hash(i + 9) * 70;
      v.rect(x, 330 - h, 46, h + 220);
      v.triangle(x - 4, 330 - h, x + 50, 330 - h, x + 23, 300 - h);
    }
  });
  v.noStroke();
  v.fill(v.linearGradient(0, 150, 0, 330, [
    [0, "rgba(255,255,255,0)"],
    [1, "rgba(255,255,255,0.45)"],
  ]));
  v.rect(0, 150, W, 400);
}

function win(v: View, x: number, y: number, w: number, h: number, curtain = "#e9d6f0"): void {
  inked(v, "#cfe7f2", () => v.rect(x, y, w, h), 2.2);
  flat(v, curtain, () => {
    v.triangle(x + 2, y + 2, x + w * 0.35, y + 2, x + 2, y + h * 0.7);
    v.triangle(x + w - 2, y + 2, x + w * 0.65, y + 2, x + w - 2, y + h * 0.7);
  });
  v.noFill();
  ink(v, 1.6);
  v.line(x + w / 2, y, x + w / 2, y + h);
  v.line(x, y + h / 2, x + w, y + h / 2);
  inked(v, "#ffffff", () => v.rect(x - 4, y + h, w + 8, 5), 1.6);
}

function facade(v: View, x: number, y: number, w: number, h: number, color: string): void {
  inked(v, color, () => v.rect(x, y, w, h), 2.6);
  // shade the right side and under the eaves with wash
  v.clip([[x, y], [x + w, y], [x + w, y + h], [x, y + h]], () => {
    v.noFill();
    v.stroke("rgba(70,40,60,0.45)");
    v.brush("wash", { opacity: 0.6 });
    v.gesture({ pressure: "even", wobble: 2 });
    v.strokeWeight(26);
    v.line(x + w - 4, y, x + w - 4, y + h);
    v.strokeWeight(30);
    v.line(x, y + 6, x + w, y + 6);
  });
}

function bricks(v: View, x: number, y: number, w: number, h: number, color: string, seed: number): void {
  v.noFill();
  v.stroke(color);
  v.brush("pencil", { opacity: 0.6 });
  v.strokeWeight(2.4);
  for (let i = 0; i < Math.floor((w * h) / 2400); i += 1) {
    const bx = x + hash(seed + i) * (w - 24);
    const by = y + hash(seed + i * 3 + 1) * (h - 10);
    v.gesture({ pressure: "even", seed: seed + i });
    v.rect(bx, by, 18, 8);
  }
}

/** Pencil lines across a wall: clapboards (horizontal) or boards (vertical). */
function boards(v: View, x: number, y: number, w: number, h: number, o: { color: string; gap: number; vertical: boolean }): void {
  v.noFill();
  v.stroke(o.color);
  v.brush("pencil", { opacity: 0.55 });
  v.strokeWeight(2);
  v.gesture({ pressure: "taper", wobble: 0.8 });
  if (o.vertical) {
    for (let bx = x + o.gap; bx < x + w - 4; bx += o.gap) v.line(bx, y + 4, bx + 1, y + h - 2);
  } else {
    for (let by = y + o.gap; by < y + h - 4; by += o.gap) v.line(x + 3, by, x + w - 3, by + 1);
  }
}

function awning(v: View, x: number, y: number, w: number, a: string, b: string): void {
  const n = Math.round(w / 20);
  const sw = w / n;
  for (let i = 0; i < n; i += 1) {
    flat(v, i % 2 === 0 ? a : b, () => {
      v.polygon([[x + i * sw, y], [x + (i + 1) * sw, y], [x + (i + 1) * sw + 4, y + 26], [x + i * sw + 4, y + 26]]);
      v.circle(x + i * sw + sw / 2 + 4, y + 26, sw);
    });
  }
  v.noFill();
  ink(v, 2.2);
  v.line(x, y, x + w, y);
  v.line(x, y, x + 4, y + 26);
  v.line(x + w, y, x + w + 4, y + 26);
}

/** Warm window and lamp glows for dusk/night, for a screen-blended cache. */
export function paintLights(v: View, night: boolean): void {
  const glow = (x: number, y: number, w: number, h: number): void => {
    v.noStroke();
    v.fill(v.ellipseGradient(x, y, w, h, [
      [0, night ? "rgba(255,214,120,0.95)" : "rgba(255,200,120,0.6)"],
      [1, "rgba(255,200,120,0)"],
    ]));
    v.ellipse(x, y, w, h);
  };
  const windows: readonly Vec2[] = [
    [178, 320], [83, 492], [288, 352], [366, 352], [285, 475], [516, 270], [571, 270], [467, 350], [516, 350], [571, 430], [663, 350], [745, 350], [664, 498], [835, 418], [965, 418], [900, 200],
  ];
  for (const [x, y] of windows) glow(x, y, 90, 80);
  glow(609, 392, 160, 160);
  if (night) {
    flat(v, "rgba(255,255,230,0.9)", () => {
      for (let i = 0; i < 40; i += 1) v.circle(hash(i + 300) * W, hash(i + 400) * 180, 1.5 + hash(i + 500) * 2);
    });
    flat(v, "#fff6d0", () => v.circle(120, 70, 50));
  }
}
