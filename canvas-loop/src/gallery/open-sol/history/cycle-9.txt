import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

const P = (points: number[][]): Vec2[] => points.map(([x, y]) => [x, y]);

function blob(s: Sketch, points: number[][], color: string): void {
  s.noStroke();
  s.fill(color);
  s.curve(P(points), true);
}

function mark(s: Sketch, points: number[][], color: string, width: number, brush = "ink"): void {
  s.noFill();
  s.stroke(color);
  s.strokeWeight(width);
  s.brush(brush as "ink");
  s.gesture({ pressure: "taper", wobble: 0.5 });
  s.curve(P(points));
}

function glint(s: Sketch, x: number, y: number, r: number): void {
  s.noStroke();
  s.fill("#ffdb91");
  s.circle(x, y, r);
  mark(s, [[x-r*1.3,y],[x+r*1.3,y]], "#ffe8b4", 1.4);
  mark(s, [[x,y-r*1.3],[x,y+r*1.3]], "#ffe8b4", 1.4);
}
