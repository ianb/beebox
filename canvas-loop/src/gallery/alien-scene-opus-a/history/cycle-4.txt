// ── geometry helpers ────────────────────────────────────────────────
function ellipsePts(cx: number, cy: number, rx: number, ry: number, n = 12): Vec2[] {
  const out: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    out.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return out;
}

/** A closed outline around a centreline, `w` wide, for limbs. */
function tube(line: Pts, w0: number, w1: number): Vec2[] {
  const left: Vec2[] = [];
  const right: Vec2[] = [];
  const n = line.length;
  for (let i = 0; i < n; i += 1) {
    const p = line[i] ?? [0, 0];
    const a = line[Math.max(0, i - 1)] ?? p;
    const b = line[Math.min(n - 1, i + 1)] ?? p;
    const dx = b[0] - a[0];
    const dy = b[1] - a[1];
    const len = Math.hypot(dx, dy) || 1;
    const nx = -dy / len;
    const ny = dx / len;
    const w = (w0 + (w1 - w0) * (i / (n - 1))) / 2;
    left.push([p[0] + nx * w, p[1] + ny * w]);
    right.push([p[0] - nx * w, p[1] - ny * w]);
  }
  return [...left, ...right.reverse()];
}

function shape(s: Sketch, pts: Pts, fillColor: string, shade: () => void, weight = 3.2, rim = true): void {
  s.noStroke();
  s.fill(fillColor);
  s.curve(pts, true);
  s.clip(s.curvePath(pts, true), () => {
    s.noFill();
    shade();
    if (rim) {
      // warm rim light: the same outline shifted left, so its band falls just
      // inside the right-hand edge (the side facing the hatch)
      s.push();
      s.translate(-5, 2);
      s.brush("airbrush", { opacity: 0.85 });
      s.stroke("#ffe596");
      s.strokeWeight(9);
      s.gesture({ pressure: "even" });
      s.curve(pts, true);
      s.pop();
    }
  });
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(weight);
  s.brush("ink", { thinning: 0.75 });
  // heavier along the bottom-left (shadow side), light along the top
  s.gesture({
    pressure: [
      [0, 0.7],
      [0.25, 1],
      [0.5, 0.95],
      [0.75, 0.5],
      [1, 0.7],
    ],
    wobble: 0.5,
  });
  s.curve(pts, true);
  s.noBrush();
}

function shadeSide(s: Sketch, cx: number, cy: number, r: number): void {
  // dusk: violet shadow multiplied over the left and bottom, a warm rim of
  // hatch light on the right, a few pencil strokes in the shadow for tooth
  s.gesture({ pressure: "even" });
  s.brush("wash", { opacity: 0.85, blend: "multiply" });
  s.stroke("#7a70b4");
  s.strokeWeight(r * 0.9);
  s.curve([
    [cx - r * 1.1, cy - r],
    [cx - r * 0.95, cy + r * 0.1],
    [cx - r * 0.3, cy + r * 1.05],
    [cx + r * 0.6, cy + r * 1.1],
  ]);
  s.brush("pencil", { opacity: 0.35 });
  s.stroke("#2f5b45");
  s.strokeWeight(2);
  s.gesture({ pressure: "taper" });
  for (let i = 0; i < Math.max(3, Math.round(r / 6)); i += 1) {
    const x = cx - r * 0.95 + i * 6;
    s.line(x, cy - r * 0.2 + i * 3, x + r * 0.35, cy + r * 0.9);
  }
  s.gesture({ pressure: "even" });
  s.brush("airbrush", { opacity: 0.9 });
  s.stroke("#ffe08a");
  s.strokeWeight(Math.max(10, r * 0.45));
  s.curve([
    [cx + r * 1.0, cy - r * 0.7],
    [cx + r * 1.05, cy + r * 0.1],
    [cx + r * 0.7, cy + r * 0.95],
  ]);
}

function alien(s: Sketch): void {
  // ground shadow
  s.noStroke();
  s.fill("rgba(15,30,30,0.35)");
  s.ellipse(AX, 526, 170, 22);

  // legs, arms and body first, then the tilted head on top
  body(s);
  s.push();
  s.translate(AX, 320);
  s.rotate(-0.12);
  s.translate(-AX, -320);
  // antennae
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(4.5);
  s.brush("ink");
  s.gesture({ pressure: "taper-out" });
  s.curve(ANT_L);
  s.curve(ANT_R);
  s.noBrush();
  for (const tip of [ANT_L[2], ANT_R[2]]) {
    if (tip === undefined) continue;
    s.noStroke();
    s.fill(
      s.radialGradient(tip[0], tip[1], 26, [
        [0, "rgba(255,230,120,0.7)"],
        [1, "rgba(255,230,120,0)"],
      ]),
    );
    s.circle(tip[0], tip[1], 52);
    shape(s, ellipsePts(tip[0], tip[1], 10, 10, 8), "#f6d548", () => {
      s.brush("airbrush", { opacity: 0.8 });
      s.stroke("#fffbe0");
      s.strokeWeight(6);
      s.line(tip[0] - 3, tip[1] - 4, tip[0] - 2, tip[1] - 3);
    }, 2.6, false);
  }

  // head
  shape(s, HEAD, GREEN, () => {
    shadeSide(s, AX, 250, 90);
  }, 3.6);
  // eyes
  for (const e of [EYE_L, EYE_R]) {
    shape(s, ellipsePts(e.x, e.y, 25, 29, 10), "#fbfaf2", () => {
      s.brush("wash", { opacity: 0.4 });
      s.stroke("#b9c4d6");
      s.strokeWeight(14);
      s.curve([
        [e.x - 22, e.y - 10],
        [e.x - 10, e.y + 18],
        [e.x + 16, e.y + 22],
      ]);
    }, 3, false);
    s.noStroke();
    s.fill("#1e2a26");
    s.circle(e.x + 3, e.y + 6, 26);
    s.fill("#fefefe");
    s.circle(e.x + 8, e.y, 9);
    s.circle(e.x - 2, e.y + 12, 4);
  }
  // cheeks
  s.noFill();
  s.brush("airbrush", { opacity: 0.55 });
  s.stroke("#f08a8a");
  s.strokeWeight(20);
  s.gesture({ pressure: "even" });
  s.line(EYE_L.x - 18, 284, EYE_L.x - 6, 284);
  s.line(EYE_R.x + 6, 284, EYE_R.x + 18, 284);
  // smile
  s.brush("ink");
  s.stroke(INK);
  s.strokeWeight(3.4);
  s.gesture({ pressure: "taper" });
  s.curve([
    [AX - 20, 290],
    [AX, 300],
    [AX + 20, 290],
  ]);
  s.noBrush();
  s.pop();
}

function body(s: Sketch): void {
  // legs and feet
  for (const leg of [LEG_L, LEG_R]) {
    shape(s, leg, GREEN, () => shadeSide(s, AX, 500, 20));
  }
  for (const foot of [FOOT_L, FOOT_R]) {
    shape(s, foot, GREEN, () => shadeSide(s, AX, 520, 20));
  }
  // hanging arm, holding a flower picked from the meadow
  shape(s, ARM_DOWN, GREEN, () => shadeSide(s, AX - 70, 400, 20));
  shape(s, ellipsePts(HAND_DOWN[0], HAND_DOWN[1], 14, 14, 10), GREEN, () =>
    shadeSide(s, HAND_DOWN[0], HAND_DOWN[1], 12),
  );
  flower(s, [HAND_DOWN[0] + 2, HAND_DOWN[1] - 4]);
  // body
  shape(s, BODY, GREEN, () => {
    shadeSide(s, AX, 410, 60);
    s.brush("wash", { opacity: 0.5 });
    s.stroke("#5da956");
    s.strokeWeight(40);
    s.curve(BELLY, true);
  });
  // waving arm
  shape(s, ARM_UP, GREEN, () => shadeSide(s, AX + 100, 330, 30));
  shape(s, ellipsePts(HAND_UP[0], HAND_UP[1], 16, 16, 10), GREEN, () =>
    shadeSide(s, HAND_UP[0], HAND_UP[1], 14),
  );
  // wave lines
  s.noFill();
  s.stroke("#fff3c4");
  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "taper" });
  s.curve([
    [HAND_UP[0] + 26, HAND_UP[1] - 26],
    [HAND_UP[0] + 34, HAND_UP[1] - 8],
    [HAND_UP[0] + 30, HAND_UP[1] + 12],
  ]);
  s.curve([
    [HAND_UP[0] + 40, HAND_UP[1] - 34],
    [HAND_UP[0] + 50, HAND_UP[1] - 10],
    [HAND_UP[0] + 44, HAND_UP[1] + 18],
  ]);
  s.noBrush();

}

function grass(s: Sketch): void {
  s.noFill();
  s.brush("ink", { thinning: 0.8 });
  for (let i = 0; i < 160; i += 1) {
    const x = s.random(0, W);
    const y = s.random(500, H + 10);
    const h = s.random(8, 22) * (y / 520);
    const lean = s.random(-6, 6);
    s.stroke(s.random() < 0.5 ? "#1d3a32" : "#7aa865");
    s.strokeWeight(s.random(1.2, 2.6));
    s.gesture({ pressure: "taper-out", wobble: 0.5 });
    s.curve([
      [x, y],
      [x + lean * 0.4, y - h * 0.6],
      [x + lean, y - h],
    ]);
  }
  s.noBrush();
}

function fireflies(s: Sketch): void {
  s.noStroke();
  const spots: Vec2[] = [
    [70, 420],
    [150, 470],
    [110, 540],
    [200, 380],
    [520, 360],
    [560, 300],
    [700, 340],
    [750, 400],
    [680, 520],
    [470, 450],
    [40, 330],
  ];
  for (const [x, y] of spots) {
    s.fill(
      s.radialGradient(x, y, 14, [
        [0, "rgba(255,240,150,0.85)"],
        [1, "rgba(255,240,150,0)"],
      ]),
    );
    s.circle(x, y, 28);
    s.fill("#fff8d0");
    s.circle(x, y, 3);
  }
}

  alien(s);
  grass(s);
  fireflies(s);
}
