// The saucer's hatch is open and spilling warm light; the alien waves at us.

// ── geometry helpers ────────────────────────────────────────────────
function ellipsePts(cx: number, cy: number, rx: number, ry: number, n = 12): Vec2[] {
  const out: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    out.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return out;
}

/** A closed outline around a centreline, `w` wide, for limbs. */
function tube(line: Pts, w0: number, w1: number): Vec2[] {
  const left: Vec2[] = [];
  const right: Vec2[] = [];
  const n = line.length;
  for (let i = 0; i < n; i += 1) {
    const p = line[i] ?? [0, 0];
    const a = line[Math.max(0, i - 1)] ?? p;
    const b = line[Math.min(n - 1, i + 1)] ?? p;
    const dx = b[0] - a[0];
    const dy = b[1] - a[1];
    const len = Math.hypot(dx, dy) || 1;
    const nx = -dy / len;
    const ny = dx / len;
    const w = (w0 + (w1 - w0) * (i / (n - 1))) / 2;
    left.push([p[0] + nx * w, p[1] + ny * w]);
    right.push([p[0] - nx * w, p[1] - ny * w]);
  }
  return [...left, ...right.reverse()];
}

// ── the saucer ──────────────────────────────────────────────────────
const SHIP = { x: 620, y: 432 };

const DOME = ellipsePts(SHIP.x, SHIP.y - 30, 72, 54, 14);

export function setup(s: Sketch): void {
  s.createCanvas(W, H);
}

function sky(s: Sketch): void {
  s.background(
    s.linearGradient(0, 0, 0, 440, [
      [0, "#1a1f4a"],
      [0.45, "#4b3f86"],
      [0.8, "#c3708a"],
      [1, "#f2b07a"],
    ]),
  );
  // moon glow and moon
  s.noStroke();
  s.fill(
    s.radialGradient(140, 110, 140, [
      [0, "rgba(255,240,200,0.45)"],
      [1, "rgba(255,240,200,0)"],
    ]),
  );
  s.circle(140, 110, 280);
  s.fill("#fbf1d2");
  s.circle(140, 110, 70);
  s.clip(ellipsePts(140, 110, 35, 35, 16), () => {
    s.noFill();
    s.brush("wash", { opacity: 0.25 });
    s.stroke("#d8c9a2");
    s.strokeWeight(16);
    s.gesture({ pressure: "even" });
    s.circle(150, 118, 14);
    s.circle(128, 98, 8);
    s.noBrush();
  });
  // stars
  s.noStroke();
  for (let i = 0; i < 70; i += 1) {
    const x = s.random(0, W);
    const y = s.random(0, 300);
    const r = s.random(0.6, 2.2);
    s.fill(`rgba(255,248,225,${(0.4 + s.random(0.6)).toFixed(2)})`);
    s.circle(x, y, r * 2);
  }
  // soft clouds near the horizon
  s.noFill();
  s.brush("airbrush", { opacity: 0.35 });
  s.stroke("#f6c3a6");
  s.strokeWeight(40);
  s.gesture({ pressure: "taper" });
  s.curve([
    [380, 300],
    [480, 290],
    [600, 300],
    [720, 290],
  ]);
  s.stroke("#e19aa8");
  s.strokeWeight(30);
  s.curve([
    [40, 340],
    [160, 330],
    [280, 342],
  ]);
  s.noBrush();
}

function land(s: Sketch): void {
  // far hills
  const far: Vec2[] = [
    [0, 380],
    [120, 360],
    [260, 386],
    [420, 366],
    [560, 384],
    [700, 360],
    [800, 372],
  ];
  s.noStroke();
  s.fill("#6a5a9a");
  s.path([
    ["move", 0, H],
    ...far.map((p): ["line", number, number] => ["line", p[0], p[1]]),
    ["line", W, H],
    ["close"],
  ]);
  // meadow
  const meadowTop: Vec2[] = [
    [0, 446],
    [200, 470],
    [400, 486],
    [600, 474],
    [800, 452],
  ];
  const meadow = [
    ...s.curvePath(meadowTop),
    ["line", W, H] as const,
    ["line", 0, H] as const,
    ["close"] as const,
  ];
  s.fill(
    s.linearGradient(0, 446, 0, H, [
      [0, "#4f7f5c"],
      [1, "#24443f"],
    ]),
  );
  s.path(meadow);
  s.clip(meadow, () => {
    s.noFill();
    s.brush("bristle", { load: 0.6 });
    s.stroke("#6b9a5e");
    s.strokeWeight(30);
    s.gesture({ pressure: "taper", wobble: 3 });
    for (let i = 0; i < 12; i += 1) {
      const y = 480 + i * 10;
      const x = s.random(-60, 600);
      s.curve([
        [x, y],
        [x + 120, y - 4],
        [x + 260, y + 3],
      ]);
    }
  });
  s.noBrush();
}

function ship(s: Sketch): void {
  // glow on the ground under the ship
  s.noStroke();
  s.fill(
    s.radialGradient(SHIP.x - 60, 490, 220, [
      [0, "rgba(255,214,130,0.55)"],
      [1, "rgba(255,214,130,0)"],
    ]),
  );
  s.ellipse(SHIP.x - 60, 490, 440, 140);
  // legs
  s.stroke(INK);
  s.strokeWeight(4);
  s.brush("ink");
  s.gesture({ pressure: "even" });
  s.line(SHIP.x - 90, 452, SHIP.x - 110, 490);
  s.line(SHIP.x + 90, 452, SHIP.x + 110, 490);
  // dome
  s.fill(
    s.linearGradient(SHIP.x - 60, SHIP.y - 80, SHIP.x + 40, SHIP.y - 10, [
      [0, "#e8f6ff"],
      [1, "#7aa4c8"],
    ]),
  );
  s.noStroke();
  s.curve(DOME, true);
  s.clip(s.curvePath(DOME, true), () => {
    s.noFill();
    s.brush("airbrush", { opacity: 0.7 });
    s.stroke("#ffffff");
    s.strokeWeight(14);
    s.gesture({ pressure: "taper" });
    s.curve([
      [SHIP.x - 52, SHIP.y - 40],
      [SHIP.x - 36, SHIP.y - 66],
      [SHIP.x - 6, SHIP.y - 78],
    ]);
  });
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.6 });
  s.curve(DOME, true);
  s.noBrush();

      [0.5, "#9aa3b8"],
      [1, "#5d6582"],

  s.noFill();

  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.6 });
  s.curve(SAUCER, true);
  s.noBrush();
}

function hatch(s: Sketch): void {
  const top: Vec2 = [SHIP.x - 58, SHIP.y + 26];
  // beam of light falling from the open hatch
  s.noStroke();
  s.fill(
    s.linearGradient(0, SHIP.y + 26, 0, 520, [
      [0, "rgba(255,236,170,0.85)"],
      [1, "rgba(255,236,170,0.05)"],
    ]),
  );
  s.path([
    ["move", top[0] - 34, top[1]],
    ["line", top[0] + 34, top[1]],
    ["line", top[0] + 70, 520],
    ["line", top[0] - 110, 520],
    ["close"],
  ]);
  // glowing doorway on the underside
  s.fill("#fff4c8");
  s.ellipse(top[0], top[1], 70, 12);
  // ramp
  s.fill(
    s.linearGradient(top[0], top[1], top[0] - 90, 508, [
      [0, "#c9c4c8"],
      [1, "#8a849c"],
    ]),
  );
  const ramp: Vec2[] = [
    [top[0] - 26, top[1] + 2],
    [top[0] + 10, top[1] + 4],
    [top[0] - 70, 512],
    [top[0] - 128, 506],
  ];
  s.polygon(ramp);
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(2.6);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.4 });
  s.polygon(ramp);
  s.noBrush();
}

function shape(s: Sketch, pts: Pts, fillColor: string, shade: () => void, weight = 3.2): void {
  s.noStroke();
  s.fill(fillColor);
  s.curve(pts, true);
  s.clip(s.curvePath(pts, true), () => {
    s.noFill();
    shade();
  });
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(weight);
  s.brush("ink", { thinning: 0.5 });
  s.gesture({ pressure: "even", wobble: 0.5 });
  s.curve(pts, true);
  s.noBrush();
}

function shadeSide(s: Sketch, cx: number, cy: number, r: number): void {
  // dusk: violet shadow multiplied over the left and bottom, a warm rim of
  // hatch light on the right, a few pencil strokes in the shadow for tooth
  s.gesture({ pressure: "even" });
  s.brush("wash", { opacity: 0.6, blend: "multiply" });
  s.stroke("#8f86c2");
  s.strokeWeight(r * 0.9);
  s.curve([
    [cx - r * 1.1, cy - r],
    [cx - r * 0.95, cy + r * 0.1],
    [cx - r * 0.3, cy + r * 1.05],
    [cx + r * 0.6, cy + r * 1.1],
  ]);
  s.brush("pencil", { opacity: 0.35 });
  s.stroke("#2f5b45");
  s.strokeWeight(2);
  s.gesture({ pressure: "taper" });
  for (let i = 0; i < Math.max(3, Math.round(r / 6)); i += 1) {
    const x = cx - r * 0.95 + i * 6;
    s.line(x, cy - r * 0.2 + i * 3, x + r * 0.35, cy + r * 0.9);
  }
  s.gesture({ pressure: "even" });
  s.brush("airbrush", { opacity: 0.75 });
  s.stroke("#ffd77e");
  s.strokeWeight(Math.max(8, r * 0.35));
  s.curve([
    [cx + r * 1.0, cy - r * 0.7],
    [cx + r * 1.05, cy + r * 0.1],
    [cx + r * 0.7, cy + r * 0.95],
  ]);
}

function flower(s: Sketch, base: Vec2): void {
  const top: Vec2 = [base[0] - 18, base[1] - 56];
  s.noFill();
  s.stroke("#2f5b3a");
  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "taper-out" });
  s.curve([base, [base[0] - 4, base[1] - 30], top]);
  s.curve([
    [base[0] - 5, base[1] - 26],
    [base[0] + 6, base[1] - 34],
    [base[0] + 12, base[1] - 30],
  ]);
  s.noBrush();
  for (let i = 0; i < 6; i += 1) {
    const a = (i / 6) * Math.PI * 2 + 0.3;
    const px = top[0] + Math.cos(a) * 8;
    const py = top[1] + Math.sin(a) * 8;
    s.noStroke();
    s.fill("#f6b9cf");
    s.push();
    s.translate(px, py);
    s.rotate(a);
    s.ellipse(0, 0, 16, 9);
    s.pop();
  }
  s.fill("#f2c94c");
  s.circle(top[0], top[1], 8);
}

function alien(s: Sketch): void {
  // ground shadow
  s.noStroke();
  s.fill("rgba(15,30,30,0.35)");
  s.ellipse(AX, 526, 170, 22);

  // legs, arms and body first, then the tilted head on top
  body(s);
  s.push();
  s.translate(AX, 320);
  s.rotate(-0.12);
  s.translate(-AX, -320);
  // antennae
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(4.5);
  s.brush("ink");
  s.gesture({ pressure: "taper-out" });
  s.curve(ANT_L);
  s.curve(ANT_R);
  s.noBrush();
  for (const tip of [ANT_L[2], ANT_R[2]]) {
    if (tip === undefined) continue;
    s.noStroke();
    s.fill(
      s.radialGradient(tip[0], tip[1], 26, [
        [0, "rgba(255,230,120,0.7)"],
        [1, "rgba(255,230,120,0)"],
      ]),
    );
    s.circle(tip[0], tip[1], 52);
    shape(s, ellipsePts(tip[0], tip[1], 10, 10, 8), "#f6d548", () => {
      s.brush("airbrush", { opacity: 0.8 });
      s.stroke("#fffbe0");
      s.strokeWeight(6);
      s.line(tip[0] - 3, tip[1] - 4, tip[0] - 2, tip[1] - 3);
    }, 2.6);
  }

  // head
  shape(s, HEAD, GREEN, () => {
    shadeSide(s, AX, 250, 90);
  }, 3.6);
  // eyes
  for (const e of [EYE_L, EYE_R]) {
    shape(s, ellipsePts(e.x, e.y, 25, 29, 10), "#fbfaf2", () => {
      s.brush("wash", { opacity: 0.4 });
      s.stroke("#b9c4d6");
      s.strokeWeight(14);
      s.curve([
        [e.x - 22, e.y - 10],
        [e.x - 10, e.y + 18],
        [e.x + 16, e.y + 22],
      ]);
    }, 3);
    s.noStroke();
    s.fill("#1e2a26");
    s.circle(e.x + 3, e.y + 6, 26);
    s.fill("#fefefe");
    s.circle(e.x + 8, e.y, 9);
    s.circle(e.x - 2, e.y + 12, 4);
  }
  // cheeks
  s.noFill();
  s.brush("airbrush", { opacity: 0.55 });
  s.stroke("#f08a8a");
  s.strokeWeight(20);
  s.gesture({ pressure: "even" });
  s.line(EYE_L.x - 18, 284, EYE_L.x - 6, 284);
  s.line(EYE_R.x + 6, 284, EYE_R.x + 18, 284);
  // smile
  s.brush("ink");
  s.stroke(INK);
  s.strokeWeight(3.4);
  s.gesture({ pressure: "taper" });
  s.curve([
    [AX - 20, 290],
    [AX, 300],
    [AX + 20, 290],
  ]);
  s.noBrush();
  s.pop();
}

function body(s: Sketch): void {
  // legs and feet
  for (const leg of [LEG_L, LEG_R]) {
    shape(s, leg, GREEN, () => shadeSide(s, AX, 500, 20));
  }
  for (const foot of [FOOT_L, FOOT_R]) {
    shape(s, foot, GREEN, () => shadeSide(s, AX, 520, 20));
  }
  // hanging arm, holding a flower picked from the meadow
  flower(s, [HAND_DOWN[0] + 2, HAND_DOWN[1] - 4]);
  shape(s, ARM_DOWN, GREEN, () => shadeSide(s, AX - 70, 400, 20));
  shape(s, ellipsePts(HAND_DOWN[0], HAND_DOWN[1], 14, 14, 10), GREEN, () =>
    shadeSide(s, HAND_DOWN[0], HAND_DOWN[1], 12),
  );
  // body
  shape(s, BODY, GREEN, () => {
    shadeSide(s, AX, 410, 60);
    s.brush("wash", { opacity: 0.5 });
    s.stroke("#5da956");
    s.strokeWeight(40);
    s.curve(BELLY, true);
  });
  // waving arm
  shape(s, ARM_UP, GREEN, () => shadeSide(s, AX + 100, 330, 30));
  shape(s, ellipsePts(HAND_UP[0], HAND_UP[1], 16, 16, 10), GREEN, () =>
    shadeSide(s, HAND_UP[0], HAND_UP[1], 14),
  );
  // wave lines
  s.noFill();
  s.stroke("#fff3c4");
  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "taper" });
  s.curve([
    [HAND_UP[0] + 26, HAND_UP[1] - 26],
    [HAND_UP[0] + 34, HAND_UP[1] - 8],
    [HAND_UP[0] + 30, HAND_UP[1] + 12],
  ]);
  s.curve([
    [HAND_UP[0] + 40, HAND_UP[1] - 34],
    [HAND_UP[0] + 50, HAND_UP[1] - 10],
    [HAND_UP[0] + 44, HAND_UP[1] + 18],
  ]);
  s.noBrush();

}

function grass(s: Sketch): void {
  s.noFill();
  s.brush("ink", { thinning: 0.8 });
  for (let i = 0; i < 160; i += 1) {
    const x = s.random(0, W);
    const y = s.random(500, H + 10);
    const h = s.random(8, 22) * (y / 520);
    const lean = s.random(-6, 6);
    s.stroke(s.random() < 0.5 ? "#1d3a32" : "#7aa865");
    s.strokeWeight(s.random(1.2, 2.6));
    s.gesture({ pressure: "taper-out", wobble: 0.5 });
    s.curve([
      [x, y],
      [x + lean * 0.4, y - h * 0.6],
      [x + lean, y - h],
    ]);
  }
  s.noBrush();
}

function fireflies(s: Sketch): void {
  s.noStroke();
  for (let i = 0; i < 14; i += 1) {
    const x = s.random(40, W - 40);
    const y = s.random(330, 560);
    s.fill(
      s.radialGradient(x, y, 14, [
        [0, "rgba(255,240,150,0.85)"],
        [1, "rgba(255,240,150,0)"],
      ]),
    );
    s.circle(x, y, 28);
    s.fill("#fff8d0");
    s.circle(x, y, 3);
  }
}

export function draw(s: Sketch): void {
  sky(s);
  land(s);
  ship(s);
  hatch(s);
  alien(s);
  grass(s);
  fireflies(s);
}
