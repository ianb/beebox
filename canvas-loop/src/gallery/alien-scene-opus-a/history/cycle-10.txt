// ── geometry helpers ────────────────────────────────────────────────
function ellipsePts(cx: number, cy: number, rx: number, ry: number, n = 12): Vec2[] {
  const out: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    out.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return out;
}

/** A closed outline around a centreline, `w` wide, for limbs. */
function tube(line: Pts, w0: number, w1: number): Vec2[] {
  const left: Vec2[] = [];
  const right: Vec2[] = [];
  const n = line.length;
  for (let i = 0; i < n; i += 1) {
    const p = line[i] ?? [0, 0];
    const a = line[Math.max(0, i - 1)] ?? p;
    const b = line[Math.min(n - 1, i + 1)] ?? p;
    const dx = b[0] - a[0];
    const dy = b[1] - a[1];
    const len = Math.hypot(dx, dy) || 1;
    const nx = -dy / len;
    const ny = dx / len;
    const w = (w0 + (w1 - w0) * (i / (n - 1))) / 2;
    left.push([p[0] + nx * w, p[1] + ny * w]);
    right.push([p[0] - nx * w, p[1] - ny * w]);
  }
  return [...left, ...right.reverse()];
}

function shape(s: Sketch, pts: Pts, fillColor: string, shade: () => void, weight = 3.2, rim = true): void {
  s.noStroke();
  s.fill(fillColor);
  s.curve(pts, true);
  s.clip(s.curvePath(pts, true), () => {
    s.noFill();
    shade();
    if (rim) {
      // warm rim light: the same outline shifted left, so its band falls just
      // inside the right-hand edge (the side facing the hatch)
      s.push();
      s.translate(-5, 2);
      s.brush("airbrush", { opacity: 1 });
      s.stroke("#fff0a8");
      s.strokeWeight(11);
      s.gesture({ pressure: "even" });
      s.curve(pts, true);
      s.pop();
    }
  });
  s.noFill();
  s.stroke(ALIEN_LINE);
  s.strokeWeight(weight);
  s.brush("ink", { thinning: 0.75 });
  // heavier along the bottom-left (shadow side), light along the top
  s.gesture({
    pressure: [
      [0, 0.7],
      [0.25, 1],
      [0.5, 0.95],
      [0.75, 0.5],
      [1, 0.7],
    ],
    wobble: 0.5,
  });
  s.curve(pts, true);
  s.noBrush();
}

/**
 * Form shadow for a part: its own outline shifted up-right, so a crescent of
 * violet falls inside the bottom-left edge (away from the hatch light), plus
 * pencil contour lines along the same crescent for tooth. `r` is the part's
 * rough radius.
 */
function shadeSide(s: Sketch, pts: Pts, r: number): void {
  const d = Math.max(4, r * 0.22);
  s.gesture({ pressure: "even" });
  s.push();
  s.translate(d, -d * 0.7);
  s.brush("wash", { opacity: 0.85, blend: "multiply", grain: 0.25 });
  s.stroke("#7a70b4");
  s.strokeWeight(d * 2.4);
  s.curve(pts, true);
  s.pop();
  s.brush("pencil", { opacity: 0.3 });
  s.stroke("#2c4f45");
  s.strokeWeight(1.6);
  s.gesture({ pressure: "taper", wobble: 0.6 });
  for (let k = 1; k <= 3; k += 1) {
    s.push();
    s.translate(d * (0.25 + k * 0.18), -d * (0.2 + k * 0.12));
    s.curve(pts, true);
    s.pop();
  }
}

/** Broad bristle strokes in a lighter green that follow the form, so the
 * lit skin looks painted. */
function skin(s: Sketch, strokes: readonly Pts[]): void {
  s.brush("bristle", { load: 0.7, opacity: 0.55 });
  s.stroke("#b4e488");
  s.strokeWeight(26);
  s.gesture({ pressure: "taper", wobble: 1.5 });
  for (const st of strokes) s.curve(st);
}

function flower(s: Sketch, base: Vec2): void {
  const top: Vec2 = [base[0] - 18, base[1] - 56];
  s.noFill();
  s.stroke("#2f5b3a");
  s.strokeWeight(3);
  s.brush("ink");
  s.gesture({ pressure: "taper-out" });
  s.curve([base, [base[0] - 4, base[1] - 30], top]);
  s.curve([
    [base[0] - 5, base[1] - 26],
    [base[0] + 6, base[1] - 34],
    [base[0] + 12, base[1] - 30],
  ]);
  s.noBrush();
  for (let i = 0; i < 6; i += 1) {
    const a = (i / 6) * Math.PI * 2 + 0.3;
    const px = top[0] + Math.cos(a) * 8;
    const py = top[1] + Math.sin(a) * 8;
    s.noStroke();
    s.fill("#f6b9cf");
    s.push();
    s.translate(px, py);
    s.rotate(a);
    s.ellipse(0, 0, 16, 9);
    s.pop();
  }
  s.fill("#f2c94c");
  s.circle(top[0], top[1], 8);
}

      s.stroke("#b9c4d6");
      s.strokeWeight(14);

function clump(s: Sketch, x: number, y: number, h: number, dark: string, light: string): void {
  const n = 5 + Math.floor(s.random(5));
  for (let i = 0; i < n; i += 1) {
    const t = (i / (n - 1)) * 2 - 1; // -1..1 across the clump
    const bh = h * (0.55 + s.random(0.45)) * (1 - Math.abs(t) * 0.35);
    const lean = t * h * 0.45 + s.random(-3, 3);
    const bx = x + t * h * 0.12;
    s.stroke(i % 2 === 0 ? dark : light);
    s.strokeWeight(Math.max(1.2, h * 0.09));
    s.gesture({ pressure: "taper-out", wobble: 0.4 });
    s.curve([
      [bx, y],
      [bx + lean * 0.3, y - bh * 0.6],
      [bx + lean, y - bh],
    ]);
  }
}

function meadowFlower(s: Sketch, x: number, y: number, r: number, color: string): void {
  s.noStroke();
  s.fill(color);
  for (let i = 0; i < 5; i += 1) {
    const a = (i / 5) * Math.PI * 2;
    s.circle(x + Math.cos(a) * r, y + Math.sin(a) * r * 0.7, r * 1.3);
  }
  s.fill("#f2c94c");
  s.circle(x, y, r * 0.9);
}

function grass(s: Sketch): void {
  s.noFill();
  s.brush("ink", { thinning: 0.8 });
  // clumps, bigger toward the bottom of the page
  for (let i = 0; i < 70; i += 1) {
    const x = s.random(-10, W + 10);
    const y = s.random(496, H + 6);
    const depth = (y - 480) / 120; // 0 far .. 1 near
    if (Math.abs(x - AX) < 64 && y < 518) continue;
    const lit = Math.hypot(x - 520, (y - 520) * 2) < 130;
    clump(s, x, y, 8 + depth * 16, lit ? "#5f7a3c" : "#183329", lit ? "#e7d98a" : "#6f9c5c");
  }
  // meadow flowers like the one in the alien's hand
  for (let i = 0; i < 26; i += 1) {
    const x = s.random(0, W);
    const y = s.random(500, H - 6);
    if (Math.abs(x - AX) < 90 && y < 540) continue;
    const r = 1.6 + ((y - 500) / 100) * 2.6;
    meadowFlower(s, x, y, r, s.random() < 0.6 ? "#f6b9cf" : "#f4efe2");
  }
  // tall dark grass framing the bottom corners
  s.noFill();
  s.brush("ink", { thinning: 0.8 });
  for (let i = 0; i < 6; i += 1) {
    clump(s, 10 + i * 22, H + 4, 70 - i * 8, "#0f241e", "#22412f");
    clump(s, W - 10 - i * 24, H + 4, 64 - i * 8, "#0f241e", "#22412f");
  }
  s.noBrush();
}

export function draw(s: Sketch): void {
  sky(s);

/** The hatch light, spilling softly over the ramp and the alien's right
 * side, screened over everything so it only ever brightens. */
function hatchBloom(s: Sketch): void {
  s.group({ opacity: 0.5, blend: "screen" }, () => {
    s.noStroke();
    s.fill(
      s.radialGradient(500, 470, 190, [
        [0, "rgba(255,200,110,0.55)"],
        [0.5, "rgba(255,190,110,0.2)"],
        [1, "rgba(255,190,110,0)"],
      ]),
    );
    s.circle(500, 470, 380);
  });
}

/** A faint warm wash over the whole page, so the paper's tooth shows
 * everywhere and the colours sit on one sheet. */
function paperTone(s: Sketch): void {
  s.group({ opacity: 0.13, blend: "multiply" }, () => {
    s.noFill();
    s.brush("wash", { opacity: 1, grain: 1 });
    s.stroke("#d8c4a4");
    s.strokeWeight(80);
    s.gesture({ pressure: "even" });
    for (let y = -20; y < H + 40; y += 60) s.line(-40, y, W + 40, y);
  });
  s.noBrush();
}
