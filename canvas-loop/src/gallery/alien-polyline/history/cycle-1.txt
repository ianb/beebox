// Friendly green alien, picture-book style. Every outline is a straight-segment
// polyline (polygon for closed shapes, move/line paths for open ones); the
// smoothness comes from sampling ellipses and Catmull-Rom splines into dense
// point lists here, in the sketch. Shading uses pencil and bristle marks clipped
// to the shapes.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

type PathCommand = Parameters<Sketch["path"]>[0][number];

const INK = "#1f2a1c";
const GREEN = "#78c96a";
const GREEN_DARK = "#4f9a45";
const GREEN_LIGHT = "#a6e08f";
const BELLY = "#5fae52";
const PAPER = "#f4f0e6";
const YELLOW = "#f3d54b";
const PINK = "#ec8f8a";

// ── geometry helpers (all produce straight-segment point lists) ──────────
/** An ellipse as a polygon, about one vertex per 3px of perimeter. */
function oval(c: Vec2, r: Vec2): Vec2[] {
  const [cx, cy] = c;
  const [rx, ry] = r;
  const n = Math.max(24, Math.round((Math.PI * (rx + ry)) / 3));
  const pts: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    pts.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return pts;
}

function at(pts: readonly Vec2[], i: number): Vec2 {
  const n = pts.length;
  const p = pts[((i % n) + n) % n];
  return p ?? [0, 0];
}

const PER_SEGMENT = 12;

function catmull(q: readonly [number, number, number, number], t: number): number {
  const [a, b, c, d] = q;
  const t2 = t * t;
  const t3 = t2 * t;
  return 0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
}

/** Sample a Catmull-Rom spline through `ctrl` into a dense polyline. */
function spline(ctrl: readonly Vec2[], closed: boolean): Vec2[] {
  const per = PER_SEGMENT;
  const out: Vec2[] = [];
  const n = ctrl.length;
  const segs = closed ? n : n - 1;
  for (let i = 0; i < segs; i += 1) {
    const p0 = closed || i > 0 ? at(ctrl, i - 1) : at(ctrl, 0);
    const p1 = at(ctrl, i);
    const p2 = at(ctrl, i + 1);
    const p3 = closed || i + 2 < n ? at(ctrl, i + 2) : at(ctrl, n - 1);
    for (let k = 0; k < per; k += 1) {
      const t = k / per;
      out.push([catmull([p0[0], p1[0], p2[0], p3[0]], t), catmull([p0[1], p1[1], p2[1], p3[1]], t)]);
    }
  }
  if (!closed) out.push(at(ctrl, n - 1));
  return out;
}

/** A closed outline around a centerline, width tapering from w0 to w1. */
function tube(center: readonly Vec2[], widths: Vec2): Vec2[] {
  const [w0, w1] = widths;
  const left: Vec2[] = [];
  const right: Vec2[] = [];
  const n = center.length;
  for (let i = 0; i < n; i += 1) {
    const a = at(center, Math.max(0, i - 1));
    const b = at(center, Math.min(n - 1, i + 1));
    const dx = b[0] - a[0];
    const dy = b[1] - a[1];
    const len = Math.hypot(dx, dy) || 1;
    const nx = -dy / len;
    const ny = dx / len;
    const w = (w0 + (w1 - w0) * (i / (n - 1))) / 2;
    const p = at(center, i);
    left.push([p[0] + nx * w, p[1] + ny * w]);
    right.push([p[0] - nx * w, p[1] - ny * w]);
  }
  return [...left, ...right.toReversed()];
}

function openPath(pts: readonly Vec2[]): PathCommand[] {
  return pts.map((p, i): PathCommand => (i === 0 ? ["move", p[0], p[1]] : ["line", p[0], p[1]]));
}

// ── drawing ──────────────────────────────────────────────────────────────
interface InkShape {
  pts: readonly Vec2[];
  fill: string;
  weight?: number;
}

/** Flat fill plus an ink outline, one polygon call. */
function inkShape(s: Sketch, o: InkShape): void {
  s.fill(o.fill);
  s.stroke(INK);
  s.strokeWeight(o.weight ?? 3.2);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.6 });
  s.polygon(o.pts);
}

interface Hatch {
  from: Vec2;
  to: Vec2;
  count: number;
  step: Vec2;
}

/** A row of parallel strokes, `from`→`to` shifted by `step` each time. */
function hatch(s: Sketch, h: Hatch): void {
  for (let i = 0; i < h.count; i += 1) {
    const ox = h.step[0] * i;
    const oy = h.step[1] * i;
    s.line(h.from[0] + ox, h.from[1] + oy, h.to[0] + ox, h.to[1] + oy);
  }
}

export function setup(s: Sketch): void {
  s.createCanvas(400, 500);
}

export function draw(s: Sketch): void {
  s.background(PAPER);

  // ground shadow
  s.noStroke();
  s.fill("#d9d3c2");
  s.polygon(oval([200, 450], [96, 14]));

  // antennae (behind the head)
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(4);
  s.brush("ink");
  s.gesture({ pressure: "taper-out", wobble: 0.5 });
  s.path(openPath(ANT_L));
  s.path(openPath(ANT_R));
  inkShape(s, { pts: BALL_L, fill: YELLOW, weight: 3 });
  inkShape(s, { pts: BALL_R, fill: YELLOW, weight: 3 });

  // legs and feet
  for (const pts of [LEG_L, LEG_R, FOOT_L, FOOT_R]) inkShape(s, { pts, fill: GREEN });

  // arms behind the body
  for (const pts of [ARM_L, ARM_R, HAND_L, HAND_R]) inkShape(s, { pts, fill: GREEN });

  // body
  inkShape(s, { pts: BODY, fill: GREEN });
  s.clip(BODY, () => {
    s.noFill();
    s.stroke(GREEN_DARK);
    s.strokeWeight(5);
    s.brush("bristle", { opacity: 0.45 });
    s.gesture({ pressure: "taper", wobble: 1.5 });
    hatch(s, { from: [236, 270], to: [246, 392], count: 4, step: [6, 0] });
  });
  s.noStroke();
  s.fill(BELLY);
  s.polygon(BELLY_SHAPE);

  // head
  inkShape(s, { pts: HEAD, fill: GREEN, weight: 3.5 });
  s.clip(HEAD, () => {
    s.noFill();
    s.stroke(GREEN_DARK);
    s.strokeWeight(2.5);
    s.brush("pencil", { opacity: 0.5 });
    s.gesture({ pressure: "taper", wobble: 1 });
    hatch(s, { from: [150, 250], to: [300, 160], count: 14, step: [6, 3] });
    s.stroke(GREEN_LIGHT);
    s.brush("bristle", { opacity: 0.6 });
    s.strokeWeight(10);
    s.path(
      openPath(
        spline(
          [
            [130, 130],
            [170, 106],
            [215, 100],
          ],
          false,
        ),
      ),
    );
  });

  // eyes
  inkShape(s, { pts: EYE_L, fill: "#ffffff", weight: 3 });
  inkShape(s, { pts: EYE_R, fill: "#ffffff", weight: 3 });
  s.noBrush();
  s.noStroke();
  s.fill(INK);
  s.circle(165, 172, 24);
  s.circle(235, 172, 24);
  s.fill("#ffffff");
  s.circle(171, 166, 8);
  s.circle(241, 166, 8);

  // cheeks
  s.fill(PINK);
  s.polygon(oval([132, 205], [11, 5.5]));
  s.polygon(oval([268, 205], [11, 5.5]));

  // smile
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(3.5);
  s.brush("ink");
  s.gesture({ pressure: "taper", wobble: 0.3 });
  s.path(openPath(SMILE));
  s.noBrush();
}
