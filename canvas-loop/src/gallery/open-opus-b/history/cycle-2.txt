// ===== sketch.ts =====
// "Old Pell Lights the Lamps" — a picture-book page. Every dusk an old heron
// walks the long boardwalk out from the stilt village, lighting the lamps one
// by one so the fishing boats can find their way home. This page: the last
// lamp, at the very end of the pier, with one small boat still out on the water.
import type { Sketch } from "@ianbicking/canvas-loop";

import { drawHeron } from "./heron.js";
import { drawLantern, drawWalkway, LAMP, LIT_LAMPS } from "./pier.js";
import { drawSky, drawVillage, drawWater, HORIZON } from "./scene.js";

export function setup(s: Sketch): void {
  s.createCanvas(800, 800);
  s.paper({ scale: 3, contrast: 0.6, seed: 7 });
}

export function draw(s: Sketch): void {
  drawSky(s);
  drawWater(s);
  drawVillage(s);
  drawWalkway(s);
  for (const lamp of LIT_LAMPS) drawLantern(s, lamp, true);
  drawLantern(s, LAMP, false);
  drawHeron(s);
  s.log("horizon", HORIZON);
  s.paperTexture(0.35);
}
// ===== scene.ts =====
// Backdrop: dusk sky, moon, far hills, the stilt village, and the water.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

export const HORIZON = 470;

export const C = {
  skyTop: "#1c2150",
  skyMid: "#4a3f7d",
  skyRose: "#b8648a",
  skyGlow: "#f3a66c",
  skyLow: "#f8cf8f",
  hill: "#6a5f92",
  hillFar: "#8f7aa6",
  house: "#3b2d4d",
  houseLit: "#5a4060",
  roof: "#271f3f",
  window: "#ffd27a",
  waterTop: "#e9a77c",
  waterMid: "#6b5d93",
  waterLow: "#232650",
  moon: "#fbf1d0",
} as const;

export function drawSky(s: Sketch): void {
  s.background(
    s.linearGradient(0, 0, 0, HORIZON, [
      [0, C.skyTop],
      [0.42, C.skyMid],
      [0.75, C.skyRose],
      [0.93, C.skyGlow],
      [1, C.skyLow],
    ]),
  );
  // Moon and its halo.
  s.noStroke();
  s.fill(s.radialGradient(190, 130, 110, [
    [0, "rgba(251,241,208,0.35)"],
    [1, "rgba(251,241,208,0)"],
  ]));
  s.circle(190, 130, 220);
  s.fill(C.moon);
  s.circle(190, 130, 58);
  // Stars.
  s.fill("rgba(255,244,214,0.85)");
  for (let i = 0; i < 40; i += 1) {
    const x = s.random(0, 800);
    const y = s.random(0, 230);
    s.circle(x, y, s.random(1, 2.6));
  }
  // Cloud bands: soft wash strokes, rose under lavender.
  s.noFill();
  s.brush("wash", { opacity: 0.35 });
  s.gesture({ pressure: "swell", wobble: 6 });
  const bands: readonly { y: number; x0: number; x1: number; c: string; w: number }[] = [
    { y: 300, x0: 380, x1: 820, c: "#d9839a", w: 30 },
    { y: 345, x0: -20, x1: 420, c: "#e293a0", w: 26 },
    { y: 390, x0: 460, x1: 820, c: "#f6b98a", w: 22 },
    { y: 250, x0: 520, x1: 760, c: "#8a6aa8", w: 18 },
  ];
  for (const b of bands) {
    s.stroke(b.c);
    s.strokeWeight(b.w);
    s.curve([[b.x0, b.y + 6], [(b.x0 + b.x1) / 2, b.y - 4], [b.x1, b.y + 3]]);
  }
  s.noBrush();
  // Far hills behind the village.
  s.noStroke();
  s.fill(C.hillFar);
  s.curve([[-20, HORIZON], [-20, 420], [80, 405], [190, 425], [300, 440], [420, 462], [480, HORIZON]], true);
  s.fill(C.hill);
  s.curve([[-20, HORIZON], [-20, 438], [100, 432], [220, 446], [360, HORIZON]], true);
}

export function drawWater(s: Sketch): void {
  s.noStroke();
  s.fill(
    s.linearGradient(0, HORIZON, 0, 800, [
      [0, C.waterTop],
      [0.18, "#b8708e"],
      [0.45, C.waterMid],
      [1, C.waterLow],
    ]),
  );
  s.rect(0, HORIZON, 800, 330);
  // Ripples: long thin bristle strokes, light near the horizon, darker below.
  s.noFill();
  s.brush("bristle", { bristles: 6, load: 0.8, spread: 0.4, flow: 0.5 });
  for (let i = 0; i < 70; i += 1) {
    const t = i / 70;
    const y = HORIZON + 6 + t * t * 320;
    const x = s.random(-40, 760);
    const len = 40 + t * 120 + s.random(0, 60);
    s.stroke(t < 0.3 ? "#f8c99a" : t < 0.6 ? "#8d7bb0" : "#3d3f78");
    s.strokeWeight(3 + t * 6);
    s.gesture({ pressure: "taper", wobble: 1 });
    s.line(x, y, x + len, y + s.random(-1, 1));
  }
  s.noBrush();
}

interface House {
  readonly x: number;
  readonly w: number;
  readonly base: number;
  readonly h: number;
  readonly roof: number;
}

const HOUSES: readonly House[] = [
  { x: 22, w: 88, base: 448, h: 66, roof: 40 },
  { x: 118, w: 70, base: 458, h: 52, roof: 34 },
  { x: 196, w: 104, base: 476, h: 74, roof: 46 },
];

function houseShape(h: House): Vec2[] {
  const top = h.base - h.h;
  return [
    [h.x, h.base],
    [h.x, top],
    [h.x + h.w / 2, top - h.roof],
    [h.x + h.w, top],
    [h.x + h.w, h.base],
  ];
}

function drawHouse(s: Sketch, h: House): void {
  // Stilts.
  s.stroke(C.roof);
  s.strokeWeight(4);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.6 });
  for (let i = 0; i <= 3; i += 1) {
    const x = h.x + 6 + (i * (h.w - 12)) / 3;
    s.line(x, h.base, x, h.base + 26);
  }
  s.noBrush();
  s.noStroke();
  s.fill(C.house);
  s.polygon(houseShape(h));
  const top = h.base - h.h;
  s.fill(C.roof);
  s.polygon([
    [h.x - 8, top + 4],
    [h.x + h.w / 2, top - h.roof - 4],
    [h.x + h.w + 8, top + 4],
    [h.x + h.w + 8, top + 10],
    [h.x + h.w / 2, top - h.roof + 6],
    [h.x - 8, top + 10],
  ]);
  // Windows with glow.
  const wy = top + h.h * 0.3;
  for (const fx of [0.28, 0.66]) {
    const wx = h.x + h.w * fx;
    s.fill(s.radialGradient(wx, wy + 8, 30, [
      [0, "rgba(255,200,110,0.45)"],
      [1, "rgba(255,200,110,0)"],
    ]));
    s.circle(wx, wy + 8, 60);
    s.fill(C.window);
    s.rect(wx - 7, wy, 14, 17);
  }
}

export function drawVillage(s: Sketch): void {
  // Reflections first, then the houses.
  s.group({ opacity: 0.45 }, () => {
    s.noStroke();
    s.fill("#2c2448");
    for (const h of HOUSES) {
      const pts = houseShape(h).map(([x, y]): Vec2 => [x, 2 * (h.base + 26) - y]);
      s.polygon(pts);
    }
    // Break the reflections with water-coloured ripple strokes.
    s.noFill();
    s.brush("bristle", { bristles: 7, load: 0.9, flow: 0.8 });
    s.gesture({ pressure: "taper", wobble: 1.2 });
    for (let y = 486; y < 640; y += 9) {
      const x = s.random(-10, 60);
      s.stroke(y < 540 ? "#c27f92" : "#7a6aa0");
      s.strokeWeight(4 + (y - 486) / 40);
      s.line(x, y, x + 260 + s.random(0, 60), y + s.random(-1, 1));
    }
    s.noBrush();
  });
  for (const h of HOUSES) drawHouse(s, h);
  drawBoat(s);
}

/** The last boat still out, far off on the right, with its own small light. */
function drawBoat(s: Sketch): void {
  const bx = 742;
  const by = HORIZON + 4;
  s.noStroke();
  s.fill(s.radialGradient(bx + 4, by - 12, 22, [
    [0, "rgba(255,214,130,0.7)"],
    [1, "rgba(255,190,110,0)"],
  ]));
  s.circle(bx + 4, by - 12, 44);
  s.fill("#3a2c4d");
  s.polygon([[bx - 16, by - 4], [bx + 18, by - 4], [bx + 12, by + 2], [bx - 12, by + 2]]);
  s.stroke("#3a2c4d");
  s.strokeWeight(1.4);
  s.line(bx - 2, by - 4, bx - 2, by - 26);
  s.noStroke();
  s.fill("#fff0b8");
  s.circle(bx + 5, by - 12, 3.4);
}
// ===== pier.ts =====
// The boardwalk out from the village, its posts, and the lamps along it.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

const X0 = 180;
const X1 = 740;

/** Top (far) edge of the deck at x. */
export function deckTop(x: number): number {
  return 505 + ((x - X0) / (X1 - X0)) * 83;
}

/** Front (near) edge of the deck at x. */
export function deckFront(x: number): number {
  return 514 + ((x - X0) / (X1 - X0)) * 108;
}

/** Perspective scale at x: 1 at the pier's end. */
export function depth(x: number): number {
  return 0.35 + (0.65 * (x - X0)) / (X1 - X0);
}

export interface Lamp {
  readonly x: number;
  readonly height: number;
  /** Where the lantern hangs. */
  readonly hx: number;
  readonly hy: number;
  readonly k: number;
}

function lampAt(x: number, height: number): Lamp {
  const k = depth(x);
  const top = deckTop(x) - height;
  return { x, height, hx: x - 34 * k, hy: top + 8 * k, k };
}

export const LAMP = lampAt(715, 285);
export const LIT_LAMPS: readonly Lamp[] = [lampAt(300, 285 * depth(300)), lampAt(430, 285 * depth(430))];

const WOOD = { deck: "#8c6b70", deckLit: "#c4917a", beam: "#3f2b38", post: "#2e2131", plank: "#5a4150" } as const;

export function drawWalkway(s: Sketch): void {
  // Posts down into the water, with short dark reflections.
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.8 });
  for (let x = X0 + 30; x <= X1; x += 62) {
    const k = depth(x);
    const y0 = deckFront(x);
    s.stroke(WOOD.post);
    s.strokeWeight(9 * k);
    s.line(x, y0, x, y0 + 50 * k);
    s.stroke("rgba(30,24,50,0.45)");
    s.strokeWeight(7 * k);
    s.line(x, y0 + 50 * k, x + 1, y0 + 90 * k);
  }
  s.noBrush();
  // Deck surface.
  s.noStroke();
  const deck: Vec2[] = [
    [X0, deckTop(X0)],
    [X1, deckTop(X1)],
    [X1, deckFront(X1)],
    [X0, deckFront(X0)],
  ];
  s.fill(s.linearGradient(X0, 0, X1, 0, [
    [0, WOOD.deck],
    [1, WOOD.deckLit],
  ]));
  s.polygon(deck);
  // Side beam under the front edge.
  s.fill(WOOD.beam);
  s.polygon([
    [X0, deckFront(X0)],
    [X1, deckFront(X1)],
    [X1, deckFront(X1) + 16],
    [X0, deckFront(X0) + 5],
  ]);
  // Plank seams.
  s.clip(deck, () => {
    s.noFill();
    s.stroke(WOOD.plank);
    s.brush("pencil", { opacity: 0.8 });
    s.gesture({ pressure: "even", wobble: 0.4 });
    for (let x = X0 + 8; x < X1; x += 5 + 16 * depth(x)) {
      s.strokeWeight(1.5 + 1.5 * depth(x));
      s.line(x, deckTop(x) - 2, x + 6 * depth(x), deckFront(x) + 2);
    }
    s.noBrush();
  });
}

export function drawLantern(s: Sketch, lamp: Lamp, _justLit: boolean): void {
  const { x, hx, hy, k } = lamp;
  const base = deckTop(x);
  const top = base - lamp.height;
  // Reflection of the light in the water.
  s.noFill();
  s.brush("wash", { opacity: 0.5 });
  s.gesture({ pressure: "taper-out", wobble: 4 });
  s.stroke("#ffcf7a");
  s.strokeWeight(26 * k);
  s.line(hx, deckFront(x) + 24 * k, hx + 2, deckFront(x) + 170 * k);
  s.noBrush();
  // Glow.
  s.noStroke();
  s.fill(s.radialGradient(hx, hy + 26 * k, 150 * k, [
    [0, "rgba(255,214,130,0.75)"],
    [0.3, "rgba(255,190,110,0.3)"],
    [1, "rgba(255,170,100,0)"],
  ]));
  s.circle(hx, hy + 26 * k, 300 * k);
  // Post and arm.
  s.stroke(WOOD.post);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.5 });
  s.strokeWeight(8 * k);
  s.line(x, base + 2, x, top);
  s.strokeWeight(4 * k);
  s.line(x + 3 * k, top + 2 * k, hx - 4 * k, top + 2 * k);
  s.strokeWeight(2 * k);
  s.line(hx, top + 2 * k, hx, hy + 6 * k);
  s.noBrush();
  // Lantern: cap, glass, base.
  s.noStroke();
  s.fill("#ffe2a0");
  s.polygon([
    [hx - 11 * k, hy + 14 * k],
    [hx + 11 * k, hy + 14 * k],
    [hx + 9 * k, hy + 40 * k],
    [hx - 9 * k, hy + 40 * k],
  ]);
  s.fill("#fff6d6");
  s.ellipse(hx, hy + 30 * k, 8 * k, 14 * k);
  s.fill("#2a1e2e");
  s.triangle(hx - 15 * k, hy + 15 * k, hx + 15 * k, hy + 15 * k, hx, hy + 4 * k);
  s.rect(hx - 11 * k, hy + 39 * k, 22 * k, 5 * k);
  s.stroke("#2a1e2e");
  s.strokeWeight(1.6 * k);
  s.line(hx, hy + 15 * k, hx, hy + 40 * k);
}
// ===== heron.ts =====
// Old Pell the heron, in a red knitted scarf, reaching the lighting pole up to
// the last lamp. Local coordinates: origin between the feet, facing +x.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

import { deckFront, deckTop, LAMP } from "./pier.js";

const FEET_X = 560;
const FEET_Y = (deckTop(FEET_X) + deckFront(FEET_X)) / 2 - 2;
const SCALE = 1.2;

const H = {
  body: "#8593ae",
  bodyShade: "#4e587a",
  belly: "#d3d8e2",
  wing: "#5b6788",
  wingDark: "#363e5e",
  rim: "#ffc98c",
  ink: "#1d1c2c",
  leg: "#c99e55",
  bill: "#e3ad45",
  scarf: "#c63a36",
  scarfDark: "#7e1f2c",
  pole: "#4a2f24",
} as const;

const BODY: readonly Vec2[] = [
  [52, -150],
  [48, -110],
  [22, -84],
  [-28, -78],
  [-84, -66],
  [-68, -116],
  [-30, -158],
  [16, -172],
];

const WING: readonly Vec2[] = [
  [30, -148],
  [32, -114],
  [6, -92],
  [-60, -76],
  [-104, -62],
  [-64, -104],
  [-18, -150],
];

const NECK: readonly Vec2[] = [
  [30, -164],
  [50, -196],
  [36, -226],
  [40, -250],
  [58, -262],
];

const HEAD = { x: 64, y: -266 } as const;
const GRIP: Vec2 = [50, -116];

type Tint = (pts: readonly Vec2[]) => void;

/** Fill a shape, then inside it a warm rim on the lamp side and a cool shade on the far side. */
function litShape(s: Sketch, pts: readonly Vec2[], base: string, extra?: Tint): void {
  s.noStroke();
  s.fill(H.rim);
  s.curve(pts, true);
  s.clip(s.curvePath(pts, true), () => {
    s.fill(base);
    s.curve(s.offset(pts, -3).map(([x, y]): Vec2 => [x - 6, y + 5]), true);
    s.fill(H.bodyShade);
    s.group({ opacity: 0.55 }, () => {
      s.curve(s.offset(pts, -2).map(([x, y]): Vec2 => [x - 26, y + 22]), true);
    });
    if (extra) extra(pts);
  });
}

export function drawHeron(s: Sketch): void {
  s.push();
  s.translate(FEET_X, FEET_Y);
  s.scale(SCALE);
  drawLegs(s);
  drawPole(s);
  drawBody(s);
  drawNeckAndHead(s);
  drawScarf(s);
  drawGrip(s);
  s.pop();
}

function drawLegs(s: Sketch): void {
  s.noFill();
  s.stroke(H.leg);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.3 });
  s.strokeWeight(4);
  s.curve([[-8, -84], [-4, -40], [-10, 0]]);
  s.curve([[8, -84], [18, -44], [14, -2]]);
  s.strokeWeight(2.5);
  s.gesture({ pressure: "taper-out" });
  for (const [fx, fy] of [[-10, 0], [14, -2]] as const) {
    s.line(fx, fy, fx + 16, fy + 3);
    s.line(fx, fy, fx - 12, fy + 2);
    s.line(fx, fy, fx + 6, fy + 6);
  }
  s.noBrush();
}

function drawBody(s: Sketch): void {
  litShape(s, BODY, H.body, () => {
    s.fill(H.belly);
    s.curve([[52, -150], [44, -104], [26, -90], [30, -130], [40, -158]], true);
  });
  litShape(s, WING, H.wing, () => {
    s.noFill();
    s.stroke(H.wingDark);
    s.brush("bristle", { bristles: 9, load: 0.9, flow: 0.7 });
    s.strokeWeight(9);
    s.gesture({ pressure: "taper", wobble: 1 });
    for (let i = 0; i < 6; i += 1) {
      s.line(16 - i * 8, -136 + i * 7, -98 + i * 6, -64 + i * 2);
    }
    s.noBrush();
  });
  s.noFill();
  s.stroke(H.ink);
  s.strokeWeight(2.4);
  s.brush("ink");
  s.gesture({ pressure: "taper", wobble: 0.6 });
  s.curve(BODY, true);
  s.curve(WING, true);
  s.noBrush();
}

function drawNeckAndHead(s: Sketch): void {
  const neck = s.tube(NECK, { width: 19, profile: [[0, 1.4], [1, 0.8]] });
  s.noStroke();
  s.fill(H.rim);
  s.path(neck);
  s.clip(neck, () => {
    s.fill(H.belly);
    s.path(s.tube(NECK.map(([x, y]): Vec2 => [x - 3, y + 1]), { width: 19, profile: [[0, 1.4], [1, 0.8]] }));
    s.fill(H.body);
    s.path(s.tube(NECK.map(([x, y]): Vec2 => [x - 13, y + 1]), { width: 19, profile: [[0, 1.4], [1, 0.8]] }));
  });
  // Head.
  const head: Vec2[] = [
    [HEAD.x + 18, HEAD.y - 1],
    [HEAD.x + 4, HEAD.y - 11],
    [HEAD.x - 14, HEAD.y - 6],
    [HEAD.x - 12, HEAD.y + 8],
    [HEAD.x + 8, HEAD.y + 10],
  ];
  litShape(s, head, H.body, () => {
    s.fill(H.belly);
    s.ellipse(HEAD.x + 6, HEAD.y + 6, 22, 9);
  });
  // Bill.
  s.noStroke();
  s.fill(H.bill);
  s.polygon([
    [HEAD.x + 14, HEAD.y - 6],
    [HEAD.x + 60, HEAD.y - 11],
    [HEAD.x + 15, HEAD.y + 5],
  ]);
  s.fill("#f6d27e");
  s.polygon([
    [HEAD.x + 14, HEAD.y - 6],
    [HEAD.x + 60, HEAD.y - 11],
    [HEAD.x + 15, HEAD.y - 1],
  ]);
  s.noFill();
  s.stroke(H.ink);
  s.brush("ink");
  s.gesture({ pressure: "taper-out", wobble: 0.5 });
  // Black crest stripe and plume.
  s.strokeWeight(5);
  s.curve([[HEAD.x + 8, HEAD.y - 9], [HEAD.x - 10, HEAD.y - 9], [HEAD.x - 30, HEAD.y - 3], [HEAD.x - 46, HEAD.y + 8]]);
  s.strokeWeight(3);
  s.curve([[HEAD.x - 4, HEAD.y - 6], [HEAD.x - 22, HEAD.y + 2], [HEAD.x - 38, HEAD.y + 18]]);
  s.gesture({ pressure: "taper", wobble: 0.5 });
  s.strokeWeight(1.6);
  s.line(HEAD.x + 15, HEAD.y - 1, HEAD.x + 52, HEAD.y - 8);
  s.strokeWeight(2.2);
  s.curve(NECK.map(([x, y]): Vec2 => [x - 9, y]));
  s.curve(head, true);
  s.noBrush();
  // Eye.
  s.noStroke();
  s.fill("#fff2c8");
  s.circle(HEAD.x + 6, HEAD.y - 2, 7);
  s.fill(H.ink);
  s.circle(HEAD.x + 7, HEAD.y - 2, 3.5);
}

function drawScarf(s: Sketch): void {
  const wrap: Vec2[] = [[12, -170], [32, -174], [50, -166]];
  const tail: Vec2[] = [[16, -168], [-8, -158], [-34, -170], [-62, -160]];
  const shapes = [s.tube(wrap, { width: 15 }), s.tube(tail, { width: 13, profile: [[0, 1], [1, 0.8]] })];
  s.noStroke();
  s.fill(H.scarf);
  for (const sh of shapes) s.path(sh);
  s.clip(shapes, () => {
    // Knit ribs and a shadow along the lower edge.
    s.noFill();
    s.stroke(H.scarfDark);
    s.brush("pencil", { opacity: 0.9 });
    s.gesture({ pressure: "even", wobble: 0.3 });
    s.strokeWeight(1.6);
    for (let x = -70; x < 60; x += 4.5) s.line(x, -186, x - 3, -150);
    s.brush("wash", { opacity: 0.6 });
    s.strokeWeight(8);
    s.curve(tail.map(([x, y]): Vec2 => [x, y + 6]));
    s.curve(wrap.map(([x, y]): Vec2 => [x, y + 7]));
    s.noBrush();
  });
  // Fringe.
  s.noFill();
  s.stroke(H.scarf);
  s.brush("ink");
  s.gesture({ pressure: "taper-out", wobble: 0.6 });
  s.strokeWeight(2.4);
  for (let i = 0; i < 4; i += 1) s.line(-60, -166 + i * 4, -74, -164 + i * 6);
  s.stroke(H.ink);
  s.strokeWeight(1.8);
  s.gesture({ pressure: "taper", wobble: 0.5 });
  for (const sh of shapes) s.path(sh);
  s.noBrush();
}

function tip(): Vec2 {
  return [(LAMP.hx - 2 - FEET_X) / SCALE, (LAMP.hy + 48 - FEET_Y) / SCALE];
}

function drawPole(s: Sketch): void {
  const [tx, ty] = tip();
  const dx = tx - GRIP[0];
  const dy = ty - GRIP[1];
  const back: Vec2 = [GRIP[0] - dx * 0.45, GRIP[1] - dy * 0.45];
  s.noFill();
  s.stroke(H.pole);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.4 });
  s.strokeWeight(3.6);
  s.line(back[0], back[1], tx, ty);
  s.noBrush();
  // Taper flame.
  s.noStroke();
  s.fill(s.radialGradient(tx, ty - 4, 18, [
    [0, "rgba(255,240,180,0.95)"],
    [1, "rgba(255,160,80,0)"],
  ]));
  s.circle(tx, ty - 4, 36);
  s.fill("#fff3c0");
  s.ellipse(tx, ty - 6, 5, 11);
}

function drawGrip(s: Sketch): void {
  // The wing's primary tips curled round the pole.
  const g: Vec2[] = [
    [GRIP[0] - 22, GRIP[1] - 8],
    [GRIP[0] + 6, GRIP[1] - 10],
    [GRIP[0] + 12, GRIP[1] + 2],
    [GRIP[0] - 4, GRIP[1] + 9],
    [GRIP[0] - 24, GRIP[1] + 6],
  ];
  litShape(s, g, H.wing);
  s.noFill();
  s.stroke(H.ink);
  s.brush("ink");
  s.gesture({ pressure: "taper", wobble: 0.4 });
  s.strokeWeight(2);
  s.curve(g, true);
  s.line(GRIP[0] - 6, GRIP[1] - 8, GRIP[0] + 4, GRIP[1] + 6);
  s.noBrush();
}
