// ===== sketch.ts =====
// "Old Pell Lights the Lamps" — a picture-book page. Every dusk an old heron
// walks the long boardwalk out from the stilt village, lighting the lamps one
// by one so the fishing boats can find their way home. This page: the last
// lamp, at the very end of the pier, with one small boat still out on the water.
import type { Sketch } from "@ianbicking/canvas-loop";

import { drawHeron } from "./heron.js";
import { drawLantern, drawWalkway, LAMP, LIT_LAMPS } from "./pier.js";
import { drawSky, drawVillage, drawWater, HORIZON } from "./scene.js";

export function setup(s: Sketch): void {
  s.createCanvas(800, 800);
  s.paper({ scale: 3, contrast: 0.6, seed: 7 });
}

export function draw(s: Sketch): void {
  drawSky(s);
  drawWater(s);
  drawVillage(s);
  drawWalkway(s);
  for (const lamp of LIT_LAMPS) drawLantern(s, lamp, true);
  drawLantern(s, LAMP, false);
  drawHeron(s);
  s.log("horizon", HORIZON);
  s.paperTexture(0.35);
}
// ===== scene.ts =====
// Backdrop: dusk sky, moon, far hills, the stilt village, and the water.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

export const HORIZON = 470;

export const C = {
  skyTop: "#1c2150",
  skyMid: "#4a3f7d",
  skyRose: "#b8648a",
  skyGlow: "#f3a66c",
  skyLow: "#f8cf8f",
  hill: "#6a5f92",
  hillFar: "#8f7aa6",
  house: "#3b2d4d",
  houseLit: "#5a4060",
  roof: "#271f3f",
  window: "#ffd27a",
  waterTop: "#e9a77c",
  waterMid: "#6b5d93",
  waterLow: "#232650",
  moon: "#fbf1d0",
} as const;

export function drawSky(s: Sketch): void {
  s.background(
    s.linearGradient(0, 0, 0, HORIZON, [
      [0, C.skyTop],
      [0.42, C.skyMid],
      [0.75, C.skyRose],
      [0.93, C.skyGlow],
      [1, C.skyLow],
    ]),
  );
  // Moon and its halo.
  s.noStroke();
  s.fill(s.radialGradient(190, 130, 110, [
    [0, "rgba(251,241,208,0.35)"],
    [1, "rgba(251,241,208,0)"],
  ]));
  s.circle(190, 130, 220);
  s.fill(C.moon);
  s.circle(190, 130, 58);
  // Stars.
  s.fill("rgba(255,244,214,0.85)");
  for (let i = 0; i < 40; i += 1) {
    const x = s.random(0, 800);
    const y = s.random(0, 230);
    s.circle(x, y, s.random(1, 2.6));
  }
  // Cloud bands: soft wash strokes, rose under lavender.
  s.noFill();
  s.brush("wash", { opacity: 0.35 });
  s.gesture({ pressure: "swell", wobble: 6 });
  const bands: readonly { y: number; x0: number; x1: number; c: string; w: number }[] = [
    { y: 300, x0: 380, x1: 820, c: "#d9839a", w: 30 },
    { y: 345, x0: -20, x1: 420, c: "#e293a0", w: 26 },
    { y: 390, x0: 460, x1: 820, c: "#f6b98a", w: 22 },
    { y: 250, x0: 520, x1: 760, c: "#8a6aa8", w: 18 },
  ];
  for (const b of bands) {
    s.stroke(b.c);
    s.strokeWeight(b.w);
    s.curve([[b.x0, b.y + 6], [(b.x0 + b.x1) / 2, b.y - 4], [b.x1, b.y + 3]]);
  }
  s.noBrush();
  // Far hills behind the village.
  s.noStroke();
  s.fill(C.hillFar);
  s.curve([[-20, HORIZON], [-20, 420], [80, 405], [190, 425], [300, 440], [420, 462], [480, HORIZON]], true);
  s.fill(C.hill);
  s.curve([[-20, HORIZON], [-20, 438], [100, 432], [220, 446], [360, HORIZON]], true);
}

export function drawWater(s: Sketch): void {
  s.noStroke();
  s.fill(
    s.linearGradient(0, HORIZON, 0, 800, [
      [0, C.waterTop],
      [0.18, "#b8708e"],
      [0.45, C.waterMid],
      [1, C.waterLow],
    ]),
  );
  s.rect(0, HORIZON, 800, 330);
  // Ripples: long thin bristle strokes, light near the horizon, darker below.
  s.noFill();
  s.brush("bristle", { bristles: 6, load: 0.8, spread: 0.4, flow: 0.5 });
  for (let i = 0; i < 70; i += 1) {
    const t = i / 70;
    const y = HORIZON + 6 + t * t * 320;
    const x = s.random(-40, 760);
    const len = 40 + t * 120 + s.random(0, 60);
    s.stroke(t < 0.3 ? "#f8c99a" : t < 0.6 ? "#8d7bb0" : "#3d3f78");
    s.strokeWeight(3 + t * 6);
    s.gesture({ pressure: "taper", wobble: 1 });
    s.line(x, y, x + len, y + s.random(-1, 1));
  }
  s.noBrush();
}

interface House {
  readonly x: number;
  readonly w: number;
  readonly base: number;
  readonly h: number;
  readonly roof: number;
}

const HOUSES: readonly House[] = [
  { x: 22, w: 88, base: 448, h: 66, roof: 40 },
  { x: 118, w: 70, base: 458, h: 52, roof: 34 },
  { x: 196, w: 104, base: 476, h: 74, roof: 46 },
];

function houseShape(h: House): Vec2[] {
  const top = h.base - h.h;
  return [
    [h.x, h.base],
    [h.x, top],
    [h.x + h.w / 2, top - h.roof],
    [h.x + h.w, top],
    [h.x + h.w, h.base],
  ];
}

function drawHouse(s: Sketch, h: House): void {
  // Stilts.
  s.stroke(C.roof);
  s.strokeWeight(4);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.6 });
  for (let i = 0; i <= 3; i += 1) {
    const x = h.x + 6 + (i * (h.w - 12)) / 3;
    s.line(x, h.base, x, h.base + 26);
  }
  s.noBrush();
  s.noStroke();
  s.fill(C.house);
  s.polygon(houseShape(h));
  const top = h.base - h.h;
  s.fill(C.roof);
  s.polygon([
    [h.x - 8, top + 4],
    [h.x + h.w / 2, top - h.roof - 4],
    [h.x + h.w + 8, top + 4],
    [h.x + h.w + 8, top + 10],
    [h.x + h.w / 2, top - h.roof + 6],
    [h.x - 8, top + 10],
  ]);
  // Windows with glow.
  const wy = top + h.h * 0.3;
  for (const fx of [0.28, 0.66]) {
    const wx = h.x + h.w * fx;
    s.fill(s.radialGradient(wx, wy + 8, 30, [
      [0, "rgba(255,200,110,0.45)"],
      [1, "rgba(255,200,110,0)"],
    ]));
    s.circle(wx, wy + 8, 60);
    s.fill(C.window);
    s.rect(wx - 7, wy, 14, 17);
  }
}

export function drawVillage(s: Sketch): void {
  // Reflections first, then the houses.
  s.group({ opacity: 0.45 }, () => {
    s.noStroke();
    s.fill("#2c2448");
    for (const h of HOUSES) {
      const pts = houseShape(h).map(([x, y]): Vec2 => [x, 2 * (h.base + 26) - y]);
      s.polygon(pts);
    }
  });
  for (const h of HOUSES) drawHouse(s, h);
}
// ===== pier.ts =====
// The boardwalk out from the village, its posts, and the lamps along it.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

const X0 = 180;
const X1 = 740;

/** Top (far) edge of the deck at x. */
export function deckTop(x: number): number {
  return 505 + ((x - X0) / (X1 - X0)) * 83;
}

/** Front (near) edge of the deck at x. */
export function deckFront(x: number): number {
  return 514 + ((x - X0) / (X1 - X0)) * 108;
}

/** Perspective scale at x: 1 at the pier's end. */
export function depth(x: number): number {
  return 0.35 + (0.65 * (x - X0)) / (X1 - X0);
}

export interface Lamp {
  readonly x: number;
  readonly height: number;
  /** Where the lantern hangs. */
  readonly hx: number;
  readonly hy: number;
  readonly k: number;
}

function lampAt(x: number, height: number): Lamp {
  const k = depth(x);
  const top = deckTop(x) - height;
  return { x, height, hx: x - 34 * k, hy: top + 8 * k, k };
}

export const LAMP = lampAt(715, 285);
export const LIT_LAMPS: readonly Lamp[] = [lampAt(330, 285 * depth(330)), lampAt(520, 285 * depth(520))];

const WOOD = { deck: "#8c6b70", deckLit: "#c4917a", beam: "#3f2b38", post: "#2e2131", plank: "#5a4150" } as const;

export function drawWalkway(s: Sketch): void {
  // Posts down into the water, with short dark reflections.
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.8 });
  for (let x = X0 + 30; x <= X1; x += 62) {
    const k = depth(x);
    const y0 = deckFront(x);
    s.stroke(WOOD.post);
    s.strokeWeight(9 * k);
    s.line(x, y0, x, y0 + 50 * k);
    s.stroke("rgba(30,24,50,0.45)");
    s.strokeWeight(7 * k);
    s.line(x, y0 + 50 * k, x + 1, y0 + 90 * k);
  }
  s.noBrush();
  // Deck surface.
  s.noStroke();
  const deck: Vec2[] = [
    [X0, deckTop(X0)],
    [X1, deckTop(X1)],
    [X1, deckFront(X1)],
    [X0, deckFront(X0)],
  ];
  s.fill(s.linearGradient(X0, 0, X1, 0, [
    [0, WOOD.deck],
    [1, WOOD.deckLit],
  ]));
  s.polygon(deck);
  // Side beam under the front edge.
  s.fill(WOOD.beam);
  s.polygon([
    [X0, deckFront(X0)],
    [X1, deckFront(X1)],
    [X1, deckFront(X1) + 16],
    [X0, deckFront(X0) + 5],
  ]);
  // Plank seams.
  s.clip(deck, () => {
    s.noFill();
    s.stroke(WOOD.plank);
    s.brush("pencil", { opacity: 0.8 });
    s.gesture({ pressure: "even", wobble: 0.4 });
    for (let x = X0 + 8; x < X1; x += 5 + 16 * depth(x)) {
      s.strokeWeight(1.5 + 1.5 * depth(x));
      s.line(x, deckTop(x) - 2, x + 6 * depth(x), deckFront(x) + 2);
    }
    s.noBrush();
  });
}

export function drawLantern(s: Sketch, lamp: Lamp, _justLit: boolean): void {
  const { x, hx, hy, k } = lamp;
  const base = deckTop(x);
  const top = base - lamp.height;
  // Reflection of the light in the water.
  s.noFill();
  s.brush("wash", { opacity: 0.5 });
  s.gesture({ pressure: "taper-out", wobble: 4 });
  s.stroke("#ffcf7a");
  s.strokeWeight(26 * k);
  s.line(hx, deckFront(x) + 24 * k, hx + 2, deckFront(x) + 170 * k);
  s.noBrush();
  // Glow.
  s.noStroke();
  s.fill(s.radialGradient(hx, hy + 26 * k, 150 * k, [
    [0, "rgba(255,214,130,0.75)"],
    [0.3, "rgba(255,190,110,0.3)"],
    [1, "rgba(255,170,100,0)"],
  ]));
  s.circle(hx, hy + 26 * k, 300 * k);
  // Post and arm.
  s.stroke(WOOD.post);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.5 });
  s.strokeWeight(8 * k);
  s.line(x, base + 2, x, top);
  s.strokeWeight(4 * k);
  s.line(x + 3 * k, top + 2 * k, hx - 4 * k, top + 2 * k);
  s.strokeWeight(2 * k);
  s.line(hx, top + 2 * k, hx, hy + 6 * k);
  s.noBrush();
  // Lantern: cap, glass, base.
  s.noStroke();
  s.fill("#ffe2a0");
  s.polygon([
    [hx - 11 * k, hy + 14 * k],
    [hx + 11 * k, hy + 14 * k],
    [hx + 9 * k, hy + 40 * k],
    [hx - 9 * k, hy + 40 * k],
  ]);
  s.fill("#fff6d6");
  s.ellipse(hx, hy + 30 * k, 8 * k, 14 * k);
  s.fill("#2a1e2e");
  s.triangle(hx - 15 * k, hy + 15 * k, hx + 15 * k, hy + 15 * k, hx, hy + 4 * k);
  s.rect(hx - 11 * k, hy + 39 * k, 22 * k, 5 * k);
  s.stroke("#2a1e2e");
  s.strokeWeight(1.6 * k);
  s.line(hx, hy + 15 * k, hx, hy + 40 * k);
}
// ===== heron.ts =====
// Old Pell the heron, in a red knitted scarf, reaching the lighting pole up to
// the last lamp. Local coordinates: origin between the feet, facing +x.
import type { Sketch, Vec2 } from "@ianbicking/canvas-loop";

import { deckFront, deckTop, LAMP } from "./pier.js";

const FEET_X = 560;
const FEET_Y = (deckTop(FEET_X) + deckFront(FEET_X)) / 2 - 4;

const H = {
  body: "#8a97b0",
  bodyShade: "#56607e",
  belly: "#cdd3df",
  wing: "#5d6987",
  wingDark: "#3c4462",
  rim: "#ffd29a",
  ink: "#1d1c2c",
  leg: "#c9a35a",
  bill: "#e0b04a",
  scarf: "#c9393a",
  scarfDark: "#8a2030",
  pole: "#5a3a2a",
} as const;

const BODY: readonly Vec2[] = [
  [42, -150],
  [38, -108],
  [12, -84],
  [-30, -82],
  [-78, -70],
  [-58, -110],
  [-22, -150],
  [12, -168],
];

const WING: readonly Vec2[] = [
  [24, -150],
  [28, -112],
  [2, -92],
  [-60, -78],
  [-96, -66],
  [-56, -104],
  [-14, -146],
];

const NECK: readonly Vec2[] = [
  [24, -160],
  [44, -192],
  [30, -222],
  [34, -246],
  [52, -258],
];

const HEAD = { x: 58, y: -262 } as const;

export function drawHeron(s: Sketch): void {
  s.push();
  s.translate(FEET_X, FEET_Y);
  drawLegs(s);
  drawBody(s);
  drawNeckAndHead(s);
  drawScarf(s);
  drawPole(s);
  s.pop();
}

function drawLegs(s: Sketch): void {
  s.noFill();
  s.stroke(H.leg);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.3 });
  s.strokeWeight(4);
  s.curve([[-6, -88], [-2, -40], [-10, 0]]);
  s.curve([[6, -88], [16, -44], [14, -2]]);
  // Toes.
  s.strokeWeight(2.5);
  s.gesture({ pressure: "taper-out" });
  for (const [fx, fy] of [[-10, 0], [14, -2]] as const) {
    s.line(fx, fy, fx + 16, fy + 3);
    s.line(fx, fy, fx - 12, fy + 2);
    s.line(fx, fy, fx + 6, fy + 6);
  }
  s.noBrush();
}

function drawBody(s: Sketch): void {
  s.noStroke();
  s.fill(H.body);
  s.curve(BODY, true);
  s.clip(s.curvePath(BODY, true), () => {
    // Belly light from the lamp, upper right.
    s.fill(H.belly);
    s.curve(s.offset(BODY, -6).map(([x, y]): Vec2 => [x + 18, y - 6]), true);
    s.fill(H.body);
    s.curve(s.offset(BODY, -4).map(([x, y]): Vec2 => [x - 8, y + 4]), true);
    s.noFill();
    s.stroke(H.bodyShade);
    s.brush("wash", { opacity: 0.6 });
    s.strokeWeight(30);
    s.curve([[-70, -76], [-30, -86], [10, -84]]);
    s.noBrush();
  });
  // Wing.
  s.fill(H.wing);
  s.curve(WING, true);
  s.clip(s.curvePath(WING, true), () => {
    s.noFill();
    s.stroke(H.wingDark);
    s.brush("bristle", { bristles: 8, load: 0.9, flow: 0.6 });
    s.strokeWeight(10);
    s.gesture({ pressure: "taper", wobble: 1 });
    for (let i = 0; i < 6; i += 1) {
      s.line(10 - i * 8, -138 + i * 6, -90 + i * 6, -70 + i * 2);
    }
    s.noBrush();
  });
  s.noFill();
  s.stroke(H.ink);
  s.strokeWeight(2.5);
  s.brush("ink");
  s.gesture({ pressure: "taper", wobble: 0.6 });
  s.curve(BODY, true);
  s.curve(WING, true);
  s.noBrush();
}

function drawNeckAndHead(s: Sketch): void {
  const neck = s.tube(NECK, { width: 18, profile: [[0, 1.4], [1, 0.8]] });
  s.noStroke();
  s.fill(H.body);
  s.path(neck);
  s.clip(neck, () => {
    s.noFill();
    s.stroke(H.belly);
    s.brush("wash", { opacity: 0.7 });
    s.strokeWeight(10);
    s.curve(NECK.map(([x, y]): Vec2 => [x + 7, y]));
    s.noBrush();
  });
  // Head.
  s.fill(H.body);
  s.ellipse(HEAD.x, HEAD.y, 32, 22);
  s.fill(H.belly);
  s.ellipse(HEAD.x + 4, HEAD.y + 4, 20, 10);
  // Bill.
  s.fill(H.bill);
  s.polygon([
    [HEAD.x + 12, HEAD.y - 5],
    [HEAD.x + 56, HEAD.y - 8],
    [HEAD.x + 13, HEAD.y + 5],
  ]);
  // Black crest stripe and plume.
  s.noFill();
  s.stroke(H.ink);
  s.brush("ink");
  s.gesture({ pressure: "taper-out", wobble: 0.5 });
  s.strokeWeight(5);
  s.curve([[HEAD.x + 6, HEAD.y - 8], [HEAD.x - 10, HEAD.y - 8], [HEAD.x - 30, HEAD.y - 2], [HEAD.x - 44, HEAD.y + 8]]);
  s.strokeWeight(3);
  s.curve([[HEAD.x - 4, HEAD.y - 6], [HEAD.x - 22, HEAD.y + 2], [HEAD.x - 38, HEAD.y + 16]]);
  s.gesture({ pressure: "taper", wobble: 0.5 });
  s.strokeWeight(2.2);
  s.curve(NECK.map(([x, y]): Vec2 => [x - 8, y]));
  s.ellipse(HEAD.x, HEAD.y, 32, 22);
  s.noBrush();
  // Eye.
  s.noStroke();
  s.fill("#fff2c8");
  s.circle(HEAD.x + 5, HEAD.y - 2, 7);
  s.fill(H.ink);
  s.circle(HEAD.x + 6, HEAD.y - 2, 3.5);
}

function drawScarf(s: Sketch): void {
  s.noFill();
  s.stroke(H.scarf);
  s.brush("bristle", { bristles: 10, load: 1, flow: 0.9 });
  s.gesture({ pressure: "even", wobble: 1 });
  s.strokeWeight(16);
  s.curve([[8, -168], [28, -172], [46, -164]]);
  s.gesture({ pressure: [[0, 1], [1, 0.6]], wobble: 2 });
  s.strokeWeight(13);
  s.curve([[12, -166], [-10, -160], [-34, -168], [-58, -158]]);
  s.stroke(H.scarfDark);
  s.strokeWeight(5);
  s.brush("ink");
  s.gesture({ pressure: "taper", wobble: 0.4 });
  s.curve([[10, -160], [-12, -154], [-36, -162], [-58, -152]]);
  s.noBrush();
}

function drawPole(s: Sketch): void {
  const tipX = LAMP.hx - FEET_X - 4;
  const tipY = LAMP.hy + 46 - FEET_Y;
  s.noFill();
  s.stroke(H.pole);
  s.brush("ink");
  s.gesture({ pressure: "even", wobble: 0.4 });
  s.strokeWeight(4);
  s.line(30, -60, tipX, tipY);
  s.noBrush();
  // Wing tip gripping the pole.
  s.noStroke();
  s.fill(H.wing);
  s.ellipse(44, -118, 22, 14);
  // Taper flame.
  s.fill(s.radialGradient(tipX, tipY - 4, 16, [
    [0, "rgba(255,240,180,0.9)"],
    [1, "rgba(255,160,80,0)"],
  ]));
  s.circle(tipX, tipY - 4, 32);
  s.fill("#fff3c0");
  s.ellipse(tipX, tipY - 5, 5, 10);
}
