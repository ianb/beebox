import type { PathCommand, Sketch, Vec2 } from "@ianbicking/canvas-loop";

// A friendly green alien, picture-book style: ink outlines drawn with s.curve,
// bristle and pencil shading clipped inside each shape.

const INK = "#1f2a1c";
const GREEN = "#79c96a";
const GREEN_DARK = "#4f9a48";
const GREEN_LIGHT = "#a6e08f";

type Pts = readonly Vec2[];

/** Points around an ellipse, for a closed curve. */
function ellipsePts(e: { cx: number; cy: number; rx: number; ry: number; n?: number }): Vec2[] {
  const { cx, cy, rx, ry } = e;
  const n = e.n ?? 10;
  const out: Vec2[] = [];
  for (let i = 0; i < n; i += 1) {
    const a = (i / n) * Math.PI * 2;
    out.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return out;
}

/** The same Catmull-Rom curve as s.curve(points, true), as path commands for clip(). */
function closedCurvePath(points: Pts): PathCommand[] {
  const n = points.length;
  const at = (i: number): Vec2 => points[(i + n) % n] ?? [0, 0];
  const first = at(0);
  const cmds: PathCommand[] = [["move", first[0], first[1]]];
  for (let i = 0; i < n; i += 1) {
    const p0 = at(i - 1);
    const p1 = at(i);
    const p2 = at(i + 1);
    const p3 = at(i + 2);
    cmds.push([
      "bezier",
      p1[0] + (p2[0] - p0[0]) / 6,
      p1[1] + (p2[1] - p0[1]) / 6,
      p2[0] - (p3[0] - p1[0]) / 6,
      p2[1] - (p3[1] - p1[1]) / 6,
      p2[0],
      p2[1],
    ]);
  }
  cmds.push(["close"]);
  return cmds;
}

const ARM_L: Pts = [

const LEG_R: Pts = [

const FOOT_R = ellipsePts({ cx: 228, cy: 443, rx: 24, ry: 11, n: 8 });
const HAND_R = ellipsePts({ cx: 330, cy: 222, rx: 14, ry: 14, n: 8 });
const HAND_L = ellipsePts({ cx: 122, cy: 380, rx: 13, ry: 13, n: 8 });

interface Part {
  readonly pts: Pts;
  readonly color?: string;
  readonly weight?: number;
  readonly closed?: boolean;
  readonly marks?: () => void;
}

function flatFill(s: Sketch, part: Part): void {
  s.noStroke();
  s.fill(part.color ?? GREEN);
  s.noBrush();
  s.curve(part.pts, true);
}

function outline(s: Sketch, part: Part): void {
  const closed = part.closed ?? true;
  s.noFill();
  s.stroke(INK);
  s.strokeWeight(part.weight ?? 3.2);
  s.brush("ink", { thinning: 0.35 });
  s.gesture({
    pressure: closed
      ? [
          [0, 0.8],
          [0.5, 1],
          [1, 0.8],
        ]
      : "taper",
    wobble: 0.6,
  });
  s.curve(part.pts, closed);
}

/** Run `part.marks` clipped to the part's outline. */
function shade(s: Sketch, part: Part): void {
  const marks = part.marks;
  if (marks === undefined) return;
  s.clip(closedCurvePath(part.pts), () => {
    s.noFill();
    marks();
  });
}

interface Stroke {
  readonly color: string;
  readonly weight: number;
  readonly pts: Pts;
  readonly opacity?: number;
}

function bristleStroke(s: Sketch, st: Stroke): void {
  s.stroke(st.color);
  s.strokeWeight(st.weight);
  s.brush("bristle", { opacity: st.opacity ?? 0.7, flow: 0.7 });
  s.gesture({ pressure: "taper", wobble: 1.5 });
  s.curve(st.pts);
}

function greenPart(s: Sketch, part: Part): void {
  flatFill(s, part);
  shade(s, part);
  outline(s, part);
}

// ── draw ─────────────────────────────────────────────────────────────
export function setup(s: Sketch): void {
  s.createCanvas(400, 500);
}

const D = GREEN_DARK;
const L = GREEN_LIGHT;

export function draw(s: Sketch): void {
  s.background("#f4f1ea");

  // ground shadow
  s.noStroke();
  s.fill("rgba(120,110,80,0.22)");
  s.ellipse(200, 450, 190, 26);

  // antennae
  for (const [base, tip] of [
    [[168, 100], [140, 42]],
    [[232, 100], [262, 36]],
  ] as const) {
    const mid: Vec2 = [(base[0] + tip[0]) / 2 + (tip[0] < 200 ? 3 : -3), (base[1] + tip[1]) / 2];
    outline(s, { pts: [base, mid, tip], weight: 3.4, closed: false });
    const ball = ellipsePts({ cx: tip[0], cy: tip[1], rx: 11, ry: 11, n: 8 });
    flatFill(s, { pts: ball, color: "#f2d34a" });
    outline(s, { pts: ball, weight: 2.8 });
  }

  // legs and feet
  for (const leg of [LEG_L, LEG_R]) {
    const x = leg[1][0] - 4;
    greenPart(s, { pts: leg, marks: () => bristleStroke(s, { color: D, weight: 9, pts: [[x, 386], [x, 438]] }) });
  }
  for (const foot of [FOOT_L, FOOT_R]) {
    const x = foot[0][0];
    greenPart(s, { pts: foot, marks: () => bristleStroke(s, { color: D, weight: 8, pts: [[x - 44, 450], [x - 20, 454], [x, 450]] }) });
  }

  // arms and hands (behind the body)
  greenPart(s, { pts: ARM_L, marks: () => bristleStroke(s, { color: D, weight: 9, pts: [[152, 304], [128, 338], [130, 374]] }) });
  greenPart(s, { pts: HAND_L, marks: () => bristleStroke(s, { color: D, weight: 8, pts: [[110, 386], [124, 392], [134, 384]] }) });
  greenPart(s, { pts: ARM_R, marks: () => bristleStroke(s, { color: D, weight: 10, pts: [[250, 292], [298, 282], [328, 258]] }) });
  greenPart(s, { pts: HAND_R, marks: () => bristleStroke(s, { color: D, weight: 8, pts: [[318, 230], [330, 236], [342, 228]] }) });

  // body
  greenPart(s, {
    pts: BODY,
    marks: () => {
      bristleStroke(s, { color: D, weight: 18, pts: [[154, 262], [146, 320], [152, 394]] });
      bristleStroke(s, { color: D, weight: 16, pts: [[156, 394], [200, 400], [248, 392]] });
      bristleStroke(s, { color: L, weight: 12, pts: [[232, 266], [242, 310], [244, 352]] });
    },
  });
  flatFill(s, { pts: BELLY, color: "#62ad56" });
  shade(s, { pts: BELLY, marks: () => bristleStroke(s, { color: D, weight: 12, pts: [[176, 326], [180, 362], [202, 380]] }) });

  // head
  greenPart(s, {
    pts: HEAD,
    marks: () => {
      bristleStroke(s, { color: D, weight: 22, pts: [[100, 160], [122, 226], [200, 262], [276, 236], [302, 176]] });
      bristleStroke(s, { color: D, weight: 12, pts: [[104, 190], [140, 236], [190, 252]] });
      bristleStroke(s, { color: L, weight: 16, pts: [[136, 114], [188, 98], [250, 106]], opacity: 0.8 });
    },
  });

  // eyes
  for (const [eye, px] of [
    [EYE_L, 165],
    [EYE_R, 235],
  ] as const) {
    flatFill(s, { pts: eye, color: "#ffffff" });
    shade(s, {
      pts: eye,
      marks: () => {
        s.stroke("#9aa6b2");
        s.strokeWeight(2.2);
        s.brush("pencil");
        s.gesture({ pressure: 0.8 });
        for (let i = 0; i < 9; i += 1) s.line(px - 34 + i * 4, 198, px - 18 + i * 4, 180);
      },
    });
    outline(s, { pts: eye, weight: 3 });
    s.noStroke();
    s.noBrush();
    s.fill(INK);
    s.circle(px, 172, 26);
    s.fill("#ffffff");
    s.circle(px + 6, 165, 8);
  }

  // cheeks: a few short pink bristle dabs
  s.group({ opacity: 0.85 }, () => {
    for (const cx of [132, 268]) {
      for (const dy of [-2, 1, 4]) {
        bristleStroke(s, {
          color: "#ee8a8a",
          weight: 6,
          pts: [
            [cx - 10, 205 + dy],
            [cx, 204 + dy],
            [cx + 10, 205 + dy],
          ],
          opacity: 1,
        });
      }
    }
  });

  // smile
  outline(s, {
    pts: [
      [179, 211],
      [188, 218],
      [200, 221],
      [212, 218],
      [221, 211],
    ],
    weight: 3.6,
    closed: false,
  });
}
