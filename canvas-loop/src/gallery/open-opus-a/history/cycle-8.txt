// ── helpers ─────────────────────────────────────────────────────────

function shift(pts: readonly Vec2[], dx: number, dy: number): Vec2[] {
  return pts.map(([x, y]): Vec2 => [x + dx, y + dy]);
}

/** Points on part of an ellipse, from angle a0 to a1 (radians). */
function ellipseArc(cx: number, cy: number, rx: number, ry: number, a0: number, a1: number, n = 12): Vec2[] {
  const pts: Vec2[] = [];
  for (let i = 0; i <= n; i += 1) {
    const a = a0 + ((a1 - a0) * i) / n;
    pts.push([cx + Math.cos(a) * rx, cy + Math.sin(a) * ry]);
  }
  return pts;
}

/** Linear interpolation through colour stops, as an rgb() string. */
function sample(stops: readonly (readonly [number, string])[], t: number, dl = 0): string {
  const rgb = (h: string): number[] => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16));
  let i = 0;
  while (i < stops.length - 2 && t > stops[i + 1][0]) i += 1;
  const [t0, c0] = stops[i];
  const [t1, c1] = stops[i + 1];
  const k = Math.min(1, Math.max(0, (t - t0) / (t1 - t0)));
  const a = rgb(c0);
  const b = rgb(c1);
  const c = a.map((v, j) => Math.round(Math.min(255, Math.max(0, v + (b[j] - v) * k + dl))));
  return `rgb(${c[0]},${c[1]},${c[2]})`;
}

/**
 * One row of horizontal paint strokes across the canvas: strokes of varied
 * length that overlap, each tapering in and out so no seam lines up.
 */
function paintRow(s: Sketch, y: number, seed: number, wobble: number): void {
  let x = -120 + ((seed * 97) % 140);
  let k = 0;
  while (x < W - 30) {
    const len = 220 + ((seed * 53 + k * 131) % 260);
    const dy = ((seed + k * 7) % 5) - 2;
    s.gesture({ pressure: [[0, 0.35], [0.18, 1], [0.8, 1], [1, 0.25]], wobble, seed: seed * 10 + k });
    s.curve([
      [x, y + dy],
      [x + len / 2, y + dy + ((k % 2) * 2 - 1) * 2],
      [x + len, y + dy + 1],
    ]);
    x += len - 60 - ((seed * 31 + k * 17) % 50);
    k += 1;
  }
}

function glow(s: Sketch, x: number, y: number, r: number, rgb: string, a: number): void {
  s.noStroke();
  s.fill(s.radialGradient(x, y, r, [[0, `rgba(${rgb},${a})`], [1, `rgba(${rgb},0)`]]));
  s.circle(x, y, r * 2);
}

// ── sky and shore ──────────────────────────────────────────────────

const SKY_STOPS: [number, string][] = [
  [0, "#181d44"],
  [0.45, "#3a3a74"],
  [0.8, "#9c7590"],
  [1, "#e3ae8a"],
];
const WATER_STOPS: [number, string][] = [
  [0, "#a98896"],
  [0.1, "#585a84"],
  [0.38, "#2b3f58"],
  [1, "#0e1c28"],
];

function sky(s: Sketch): void {
  s.noStroke();
  s.fill(s.linearGradient(0, 0, 0, HORIZON, SKY_STOPS));
  s.rect(0, 0, W, HORIZON + 4);
  // paint over the gradient in long loaded strokes, row by row
  s.noFill();
  for (let i = 0; i < 24; i += 1) {
    const y = 6 + i * 12.5;
    s.brush("bristle", { opacity: 0.6, load: 0.7 });
    s.stroke(sample(SKY_STOPS, y / HORIZON, i % 2 === 0 ? 30 : -22));
    s.strokeWeight(26);
    paintRow(s, y, i, 3);
  }
  s.brush("wash", { opacity: 0.5 });
  s.gesture({ pressure: "taper", wobble: 3 });
  s.stroke("#6a5a8c");
  s.strokeWeight(30);
  s.curve([[60, 150], [220, 140], [380, 150]]);
  s.curve([[420, 110], [600, 100], [760, 115]]);
  s.stroke("#c9919a");
  s.strokeWeight(18);
  s.curve([[0, 230], [180, 222], [330, 228]]);
  s.curve([[480, 215], [650, 210], [800, 220]]);
  s.noBrush();
  s.noStroke();
  glow(s, 612, 66, 16, "255,240,210", 0.5);
  s.fill("#fff4d8");
  s.circle(612, 66, 3.5);
}

const HILLS: Vec2[] = [
  [-20, 302],
  [-10, 268],
  [90, 252],
  [200, 270],
  [300, 286],
  [420, 292],
  [560, 284],
  [700, 266],
  [820, 272],
  [820, 302],
];

function farShore(s: Sketch): void {
  s.noStroke();
  s.fill("#3d3f66");
  s.curve(HILLS, true);
  s.fill("#2e3054");
  for (let i = 0; i < 26; i += 1) {
    const x = 20 + i * 11 + (i % 3) * 3;
    const base = 290 - Math.sin(i * 0.4) * 6 - (x < 200 ? 12 : 0);
    const h = 12 + (i % 4) * 4;
    s.triangle(x - 4, base + 4, x + 4, base + 4, x, base - h);
  }
  // one lit window: home
  s.fill("#ffcf7a");
  s.rect(146, 270, 4, 4);
  glow(s, 148, 272, 10, "255,200,110", 0.6);
}

// ── water ──────────────────────────────────────────────────────────

function water(s: Sketch): void {
  s.noStroke();
  s.fill(s.linearGradient(0, HORIZON, 0, H, WATER_STOPS));
  s.rect(0, HORIZON, W, H - HORIZON);
  s.noFill();
  for (let i = 0; i < 30; i += 1) {
    const y = HORIZON + 8 + i * 16.5;
    s.brush("bristle", { opacity: 0.45, load: 0.6 });
    s.stroke(sample(WATER_STOPS, (y - HORIZON) / (H - HORIZON), i % 2 === 0 ? 24 : -16));
    s.strokeWeight(22 + i * 0.5);
    paintRow(s, y, 100 + i, 1.5);
  }
  // the far shore mirrored, softened
  s.group({ opacity: 0.4 }, () => {
    s.fill("#3a3c62");
    s.curve(HILLS.map(([x, y]): Vec2 => [x, HORIZON + (HORIZON - y) * 0.6 + 1]), true);
  });
  // evening mist lying on the far water
  s.brush("airbrush", { opacity: 0.35 });
  s.gesture({ pressure: "taper", wobble: 4 });
  s.stroke("#c8a8b4");
  s.strokeWeight(28);
  s.curve([[-40, 300], [200, 304], [420, 298], [640, 302], [840, 298]]);
  // brushed bands: reflected sky near the horizon, dark swell nearer
  s.noFill();
  s.gesture({ pressure: "taper", wobble: 1 });
  for (let i = 0; i < 30; i += 1) {
    const t = i / 30;
    const y = HORIZON + 6 + t * t * 480;
    const x = (i * 211) % 700;
    const len = 60 + t * 220 + ((i * 37) % 80);
    s.brush("wash", { opacity: 0.25 + t * 0.15 });
    s.stroke(t < 0.25 ? "#d0a2a0" : i % 2 === 0 ? "#536c8c" : "#1a2a3a");
    s.strokeWeight(3 + t * 16);
    s.line(x - len / 2, y, x + len / 2, y + 1);
  }
  // little light glints on the near water
  s.brush("ink", { opacity: 0.5 });
  s.stroke("#8aa0bc");
  for (let i = 0; i < 8; i += 1) {
    const x = 60 + ((i * 173) % 680);
    const y = 600 + ((i * 59) % 180);
    s.strokeWeight(1.4);
    s.line(x, y, x + 18 + (i % 4) * 8, y);
  }
}

  s.group({ opacity: 0.4 }, () => {
    s.noStroke();
    s.fill("#98865a");
    s.curve(FISH_BODY, true);
    s.clip(s.curvePath(FISH_BODY, true), () => {

      s.noFill();

      s.stroke("#2e2a1e");
      s.strokeWeight(2.2);
      s.gesture({ pressure: "taper" });
      for (let r = 0; r < 7; r += 1) {
        for (let c = 0; c < 5; c += 1) {
          const cx = 596 + c * 28 + r * 9;
          const cy = 590 + r * 28;
          s.curve(ellipseArc(cx, cy, 14, 10, 0.3, Math.PI - 0.3, 6));
        }
      }
    });
    // dorsal fin, along the back
    s.fill("#2a2a20");
    s.curve([[680, 578], [740, 590], [796, 650], [778, 690], [736, 640]], true);
  });
  // deep water over the lower body
  s.noStroke();
  s.fill(s.ellipseGradient(690, 800, 360, 440, [[0, "rgba(14,28,40,0.75)"], [0.6, "rgba(14,28,40,0.4)"], [1, "rgba(14,28,40,0)"]]));

function lanternReflection(s: Sketch): void {
  glowEllipse(s, LANTERN[0] + 10, 548, 300, 90, "255,176,90", 0.3);
  s.noFill();
  s.stroke("#f4b058");
  // broken glints, wider and fainter as they come toward us
  const dxs = [0, -6, 5, -3, 8, -9, 2, 10, -12, 4, -6, 13, -2, 7];
  for (let i = 0; i < dxs.length; i += 1) {
    const y = 528 + i * 8 + i * i * 0.55;
    const half = 12 + ((i * 7) % 11) + i * 1.2;
    const x = LANTERN[0] + 4 + dxs[i];
    s.brush("ink", { opacity: 0.8 - i * 0.045, thinning: 0.8 });
    s.gesture({ pressure: "taper", wobble: 1.2, seed: 300 + i });
    s.strokeWeight(2 + (i % 2));
    s.line(x - half, y, x + half * 0.4, y + 1);
    if (i % 2 === 1) s.line(x + half * 0.6, y + 2, x + half * 1.2, y + 2);
  }
  s.brush("wash", { opacity: 0.4 });
  s.gesture({ pressure: "taper" });
  s.strokeWeight(26);
  s.line(LANTERN[0] + 4, 530, LANTERN[0] + 6, 640);
}

function glowEllipse(s: Sketch, x: number, y: number, w: number, h: number, rgb: string, a: number): void {
  s.noStroke();
  s.fill(s.ellipseGradient(x, y, w, h, [[0, `rgba(${rgb},${a})`], [1, `rgba(${rgb},0)`]]));
  s.ellipse(x, y, w, h);
}

// ── lily pads in the near water ────────────────────────────────────

function lilyPad(s: Sketch, x: number, y: number, w: number, notch: number): void {
  const h = w * 0.32;
  const pts: Vec2[] = [];
  for (let i = 0; i <= 14; i += 1) {
    const a = notch + 0.16 + (i / 14) * (Math.PI * 2 - 0.32);
    pts.push([x + Math.cos(a) * w * 0.5, y + Math.sin(a) * h * 0.5]);
  }
  pts.push([x, y]);
  s.noStroke();
  s.fill("#152a28");
  s.polygon(pts);
  s.clip(pts, () => {
    s.noFill();
    s.brush("bristle", { opacity: 0.7, load: 0.8 });
    s.gesture({ pressure: "even", wobble: 1 });
    s.stroke("#22403a");
    s.strokeWeight(h * 0.5);
    s.line(x - w * 0.5, y - h * 0.2, x + w * 0.5, y - h * 0.25);
    // the sky caught on the far rim
    s.brush("wash", { opacity: 0.55 });
    s.stroke("#6c6c92");
    s.strokeWeight(6);
    s.curve(ellipseArc(x, y, w * 0.5, h * 0.5, Math.PI * 1.1, Math.PI * 1.9));
  });
  s.noFill();
  s.brush("ink");
  s.stroke("#08120e");
  s.strokeWeight(1.8);
  s.gesture({ pressure: "swell", wobble: 0.6 });
  s.curve(ellipseArc(x, y, w * 0.5, h * 0.5, 0, Math.PI));
}

function lilyPads(s: Sketch): void {

  s.fill("#c8a8b8");
  s.curve([[289, 718], [292, 711], [295, 718]], true);
}

// ── foreground reeds ───────────────────────────────────────────────

function reeds(s: Sketch): void {
  s.noFill();
  s.brush("ink");
  s.stroke("#09121a");
  const left: Vec2[] = [];
  for (let i = 0; i < 14; i += 1) {
    const x = 10 + i * 12 + (i % 3) * 5;
    const top = 600 + ((i * 47) % 120);
    const tip: Vec2 = [x + 14 + (i % 4) * 6, top];
    left.push(tip);
    s.strokeWeight(3 + (i % 3));
    s.gesture({ pressure: "taper-out", wobble: 1.5 });
    s.curve([[x, 820], [x + 4, (820 + top) / 2], tip]);
  }
  const right: Vec2[] = [];
  for (let i = 0; i < 9; i += 1) {
    const x = 722 + i * 10;
    const top = 570 + ((i * 53) % 140);
    const tip: Vec2 = [x - 12 - (i % 3) * 5, top];
    right.push(tip);
    s.strokeWeight(3 + (i % 3));
    s.gesture({ pressure: "taper-out", wobble: 1.5 });
    s.curve([[x, 820], [x - 2, (820 + top) / 2], tip]);
  }
  // cattails on stiff stalks
  s.strokeWeight(2.5);
  s.gesture({ pressure: "even" });
  s.line(56, 820, 58, 620);
  s.line(712, 820, 706, 580);
  s.noBrush();
  s.noStroke();
  s.fill("#2a1a10");
  s.ellipse(58, 640, 10, 36);
  s.ellipse(706, 600, 10, 34);
}
